\subsubsection*{Problem Set: Vectors, Trigonometry, and Frames of Reference --- Solutions}

\begin{enumerate}

\item \textbf{Force components} ($\vec{F} = 50\,\text{N}$ at $37^\circ$ above horizontal).

\begin{enumerate}
\item $F_x = F\cos\theta = (50\,\text{N})\cos 37^\circ \approx (50)(0.799) \approx \boxed{40\,\text{N}}$

\item $F_y = F\sin\theta = (50\,\text{N})\sin 37^\circ \approx (50)(0.602) \approx \boxed{30\,\text{N}}$
\end{enumerate}

\emph{Check:} $\sqrt{40^2 + 30^2} = \sqrt{1600 + 900} = \sqrt{2500} = 50\,\text{N}.$ \checkmark

\vspace{0.8cm}

\item \textbf{Magnitude and direction} ($v_x = 12\,\text{m/s}$, $v_y = 5\,\text{m/s}$).

\begin{enumerate}
\item $|\vec{v}| = \sqrt{v_x^2 + v_y^2} = \sqrt{12^2 + 5^2} = \sqrt{144 + 25} = \sqrt{169} = \boxed{13\,\text{m/s}}$

\item $\theta = \arctan\!\left(\dfrac{v_y}{v_x}\right) = \arctan\!\left(\dfrac{5}{12}\right) \approx 22.62^\circ \approx \boxed{23^\circ}$ north of east.
\end{enumerate}

\vspace{0.8cm}

\item \textbf{Frame of reference --- train.}

\begin{enumerate}
\item Take east as positive. Passenger moves at $+1.5\,\text{m/s}$ relative to the train, which moves at $+25\,\text{m/s}$ relative to the ground.
\[
v_{\text{passenger/ground}} = 1.5 + 25 = \boxed{26.5\,\text{m/s east}}
\]

\item The car moves at $-15\,\text{m/s}$ (west) relative to the ground.
\[
v_{\text{passenger/car}} = v_{\text{passenger/ground}} - v_{\text{car/ground}} = 26.5 - (-15) = \boxed{41.5\,\text{m/s east}}
\]
\end{enumerate}

\newpage

\item \textbf{Frame of reference --- boat in river.}

Boat velocity relative to water: $(0, 4.0)\,\text{m/s}$ (north). \\
Water velocity relative to ground: $(3.0, 0)\,\text{m/s}$ (east). \\
Boat velocity relative to ground:
\[
\vec{v}_{\text{boat/ground}} = (3.0, 4.0)\,\text{m/s}.
\]

\begin{enumerate}
\item $|\vec{v}| = \sqrt{3.0^2 + 4.0^2} = \sqrt{9 + 16} = \sqrt{25} = \boxed{5.0\,\text{m/s}}$

\item Measuring east of north, the east component is opposite the angle and the north component is adjacent:
\[
\theta = \arctan\!\left(\dfrac{3.0}{4.0}\right) \approx 36.87^\circ \approx \boxed{37^\circ}\ \text{east of north.}
\]
\end{enumerate}

\vspace{0.8cm}

\item \textbf{Energy unit conversions} ($E = 1.0\times 10^{6}\,\text{J}$).

\begin{enumerate}
\item Kilocalories:
\[
E = 1.0\times 10^{6}\,\text{J} \times \dfrac{1\,\text{kcal}}{4184\,\text{J}} \approx \boxed{239\,\text{kcal}}
\]
(About 240 ``Calories'' on a food label.)

\item Kilowatt-hours:
\[
E = 1.0\times 10^{6}\,\text{J} \times \dfrac{1\,\text{kWh}}{3.6\times 10^{6}\,\text{J}} \approx \boxed{0.28\,\text{kWh}}
\]

\item Electron-volts:
\[
E = 1.0\times 10^{6}\,\text{J} \times \dfrac{1\,\text{eV}}{1.6\times 10^{-19}\,\text{J}} = \boxed{6.25\times 10^{24}\,\text{eV}}
\]
\end{enumerate}

\end{enumerate}
