\subsubsection*{2.4 Quiz: Motion and Kinematics}

\begin{enumerate}[itemsep=1cm]

\item Round each value to three significant figures.
  \begin{multicols}{2}
  \begin{enumerate}[itemsep=1cm]
    \item 0.007146 m
    \item 52.837 kg
    \item 6.0052 s
    \item 18,470 N
  \end{enumerate}
  \end{multicols}
  \vspace{1cm}

\item Perform each operation and give the answer to 3 significant figures.
  \begin{multicols}{2}
  \begin{enumerate}[itemsep=1.5cm]
    \item $(6.5\times10^{3}) + (8.4\times10^{2})$
    \item $(5.60\times10^{-2}) \times (3.0\times10^{5})$
    \item $\dfrac{9.6\times10^{7}}{4.8\times10^{4}}$
  \end{enumerate}
  \end{multicols}
  \vspace{1.5cm}

\item Convert each value. Show one line of work using unit factors.
  \begin{enumerate}[itemsep=2.5cm]
    \item $54.0~\text{km/h}$ to m/s
    \item $12.5~\text{m/s}$ to km/h
  \end{enumerate}
  \vspace{1.5cm}

\newpage
\item A student walks along a straight path (forward is $+$). Start $x_0 = 2.5$ m.
  \begin{enumerate}[itemsep=1.5cm]
    \item He walks to $x = 14.5$ m. Find displacement $\Delta x$.
    \item From $x=14.5$ m he walks back to $x=6.0$ m. Find $\Delta x$.
    \item Find total displacement from start to finish.
    \item Find total distance traveled.
  \end{enumerate}
  \vspace{2cm}

\subsubsection*{Kinematics Equations}

\textbf{Formulas:} \quad $v = v_0 + at$ \quad\quad $d = v_0 t + \frac{1}{2}at^2$ \quad\quad $v^2 = v_0^2 + 2ad$ \quad\quad $d = \frac{1}{2}(v_0 + v)t$

\vspace{1cm}

\item A jogger runs along a straight trail at an average velocity of $8.0~\text{km/h}$ for $1.5~\text{h}$. How far does she travel?
  \vspace{4cm}

\item A skateboard accelerates uniformly from rest at $1.8~\text{m/s}^2$.
  \begin{enumerate}[itemsep=3.5cm]
    \item What is its velocity after $5.0~\text{s}$?
    \item How far does it travel during this time?
  \end{enumerate}
  \vspace{2cm}

\newpage
\item A car traveling at $30~\text{m/s}$ applies its brakes and decelerates at $2.0~\text{m/s}^2$.
  \begin{enumerate}[itemsep=4cm]
    \item How long does it take for the car to stop?
    \item What distance does the car travel while braking?
  \end{enumerate}
  \vspace{2cm}

\item A ball is thrown vertically upward with an initial velocity of $20~\text{m/s}$. Use $g = 10~\text{m/s}^2$.
  \begin{enumerate}[itemsep=4cm]
    \item How long does it take to reach its maximum height?
    \item What is the maximum height reached?
    \item What is the ball's velocity when it returns to the thrower's hand?
  \end{enumerate}

\end{enumerate}
