\subsubsection*{2.3 PreQuiz: Motion and Kinematics}

\item Round each value to three significant figures.
  \begin{multicols}{2}
  \begin{enumerate}[itemsep=0.75cm]
    \item 0.005283 m
    \item 47.652 kg
    \item 8.0074 s
    \item 15,340 N
  \end{enumerate}
  \end{multicols}
  \vspace{0.75cm}

\item Write each number in proper scientific notation ($1 \le a < 10$).
  \begin{multicols}{2}
  \begin{enumerate}[itemsep=0.75cm]
    \item 0.000628
    \item 72,500
    \item 304,000,000
    \item 0.00915
  \end{enumerate}
  \end{multicols}
  \vspace{0.75cm}

\item Perform each operation and give the answer to 3 significant figures.
  \begin{multicols}{2}
  \begin{enumerate}[itemsep=1.25cm]
    \item $(4.8\times10^{2}) + (5.2\times10^{3})$
    \item $(7.20\times10^{-3}) \times (4.0\times10^{4})$
    \item $\dfrac{8.4\times10^{6}}{2.1\times10^{3}}$
  \end{enumerate}
  \end{multicols}
  \vspace{1cm}

\item Convert each value. Show one line of work using unit factors.
  \begin{multicols}{2}
  \begin{enumerate}[itemsep=2cm]
    \item $4.75~\text{m}$ to cm
    \item $2.80~\text{km}$ to m
    \item $0.450~\text{kg}$ to g
    \item $6.30\times10^{4}~\text{g}$ to kg
  \end{enumerate}
  \end{multicols}
  \vspace{1cm}

\item Convert each speed.
  \begin{enumerate}[itemsep=2cm]
    \item $72.0~\text{km/h}$ to m/s
    \item $8.50~\text{m/s}$ to km/h
  \end{enumerate}
  \vspace{1cm}

\newpage
\item A student walks along a straight hallway (forward is $+$). Start $x_0 = 3.0$ m.
  \begin{enumerate}[itemsep=1.25cm]
    \item She walks to $x = 11.0$ m. Find displacement $\Delta x$.
    \item From $x=11.0$ m she walks back to $x=5.0$ m. Find $\Delta x$.
    \item Find total displacement from start to finish.
    \item Find total distance traveled.
  \end{enumerate}
  \vspace{1cm}

\item A cart moves from $x_1 = 2.0$ m at $t_1=0$ s to $x_2 = 8.6$ m at $t_2 = 4.0$ s.
  \begin{enumerate}[itemsep=1.25cm]
    \item Find displacement.
    \item Find average velocity $v_\text{avg}$.
  \end{enumerate}
  \vspace{1cm}

\item One-dimensional vector additions: find the total displacement.
  \begin{multicols}{2}
  \begin{enumerate}[itemsep=1.5cm]
    \item $\Delta x_1 = +18$ m, $\Delta x_2 = -7$ m
    \item $+2.30$ km, $-0.80$ km, $+1.50$ km
  \end{enumerate}
  \end{multicols}
  \vspace{1.5cm}

\subsubsection*{Kinematics Equations}

\textbf{Formulas:} \quad $v = v_0 + at$ \quad\quad $d = v_0 t + \frac{1}{2}at^2$ \quad\quad $v^2 = v_0^2 + 2ad$ \quad\quad $d = \frac{1}{2}(v_0 + v)t$

\vspace{1cm}

\item How far can a cyclist travel in $2.5~\text{h}$ along a straight road if her average velocity is $18~\text{km/h}$?
  \vspace{4cm}

\item A car accelerates uniformly from rest at $2.4~\text{m/s}^2$.
  \begin{enumerate}[itemsep=3cm]
    \item What is its velocity after $6.0~\text{s}$?
    \item How far does it travel during this time?
  \end{enumerate}
  \vspace{2cm}

\newpage
\item A train traveling at $25~\text{m/s}$ applies its brakes and decelerates at $1.5~\text{m/s}^2$.
  \begin{enumerate}[itemsep=3.5cm]
    \item How long does it take for the train to stop?
    \item What distance does the train travel while braking?
  \end{enumerate}
  \vspace{2cm}

\item A ball is thrown vertically upward with an initial velocity of $15~\text{m/s}$. Use $g = 10~\text{m/s}^2$.
  \begin{enumerate}[itemsep=3.5cm]
    \item How long does it take to reach its maximum height?
    \item What is the maximum height reached?
    \item What is the ball's velocity when it returns to the thrower's hand?
  \end{enumerate}

\end{enumerate}
