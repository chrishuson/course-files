\subsubsection*{3.3 Pre-Test: Trigonometric Functions --- Markscheme}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Radian measure: degrees to radians exact; radians to degrees; arc length and sector area; angle from a given area [9 marks] ---
\item[1.] [Maximum mark: 9]

\textbf{(a)} $135^\circ = \dfrac{3\pi}{4}$ \hfill \textbf{A1}

\textbf{(b)} $2.4 \times \dfrac{180}{\pi}$ \hfill \textbf{M1}

$= 138^\circ$ \quad $(137.509\ldots)$ \hfill \textbf{A1}

\emph{A1 for 138 or the display with an ellipsis; 137.5 with no ellipsis
earns A0 (markscheme-spec \S21). Wrong method: $2.4 \times \frac{\pi}{180}
= 0.0419$ earns M0A0.}

Sector: $r = 9$, $\theta = 1.2$.

\textbf{(c)} $l = 9 \times 1.2$ \hfill \textbf{M1}

$= 10.8$ cm \hfill \textbf{A1}

\textbf{(d)} $A = \dfrac{1}{2} \times 9^{2} \times 1.2$ \hfill \textbf{M1}

$= 48.6\ \text{cm}^{2}$ \hfill \textbf{A1}

\emph{Wrong methods: the degree formula, $\frac{1.2}{360} \times 2\pi
\times 9 = 0.188$ in (c) or $\frac{1.2}{360} \times \pi \times 9^{2}$ in
(d), earns M0A0; $r$ for $r^{2}$, $\frac{1}{2} \times 9 \times 1.2 = 5.4$,
earns M0A0.}

\textbf{(e)} $60 = \dfrac{1}{2} \times 9^{2} \times \theta$ \hfill \textbf{M1}

$\theta = 1.48$ \quad $(1.48148\ldots)$ \hfill \textbf{A1}

\emph{Accept the exact value $\frac{40}{27}$. Wrong methods: the
$\frac{1}{2}$ dropped, $\frac{60}{81} = 0.741$, earns M0A0; the arc-length
formula, $\frac{60}{9} = 6.67$, earns M0A0. An answer in degrees only,
$84.9^\circ$, earns M1A0.}

%--- Problem 2: Unit circle: exact values; cos given with quadrant, find sin and tan exactly; solve 2sin x + 1 = 0 on [0, 2pi] exactly [10 marks] ---
\item[2.] [Maximum mark: 10]

\textbf{(a)(i)} $\cos\dfrac{3\pi}{4} = -\dfrac{\sqrt{2}}{2}$ \hfill \textbf{A1}

\textbf{(a)(ii)} $\sin\dfrac{5\pi}{3} = -\dfrac{\sqrt{3}}{2}$ \hfill \textbf{A1}

\textbf{(a)(iii)} $\tan\dfrac{4\pi}{3} = \sqrt{3}$ \hfill \textbf{A1}

\emph{Accept $-\frac{1}{\sqrt{2}}$ and $\frac{3}{\sqrt{3}}$. Decimals from
the GDC ($-0.707$, $-0.866$, $1.73$) earn A0: exact values are asked
for. Common errors, each A0: wrong sign in (ii) or (iii).}

\textbf{(b)(i)} $\sin^{2}\theta = 1 - \left(-\dfrac{5}{13}\right)^{2}
= \dfrac{144}{169}$ \hfill \textbf{M1}

$\sin\theta = \pm\dfrac{12}{13}$ \hfill \textbf{A1}

$\sin\theta = -\dfrac{12}{13}$, since $\theta$ is in the third quadrant
\hfill \textbf{A1}

\textbf{(b)(ii)} $\tan\theta = \dfrac{-12/13}{-5/13} = \dfrac{12}{5}$
\hfill \textbf{A1}

\emph{Accept a right-triangle method (sides 5, 12, 13) for the M1A1. The
final A1 in (i) is for the negative sign. Wrong method: the GDC route
$\sin\left(\arccos\left(-\frac{5}{13}\right)\right) = +\frac{12}{13}$
gives the second-quadrant angle, earning M1A1A0, and then
$\tan = -\frac{12}{5}$, A0 unless FT. FT (ii) from the candidate's
$\sin\theta$. Decimals ($-0.923$, $2.4$) earn A0 for exactness, except
that $2.4$ is exact: accept $2.4$ in (ii).}

\textbf{(c)} $\sin x = -\dfrac{1}{2}$ \hfill \textbf{M1}

$x = \dfrac{7\pi}{6}$, \ $x = \dfrac{11\pi}{6}$ \hfill \textbf{A1A1}

\emph{Wrong methods: the GDC value $x = -\frac{\pi}{6}$ ($-0.524$) is
outside the interval, A0; $x = \frac{\pi}{6}, \frac{5\pi}{6}$ (sign of
$\sin x$ wrong) earns M0A0A0. Decimals ($3.67$, $5.76$) earn A1A0 at most:
the question asks for multiples of $\pi$. Extra values outside the
interval earn the final A0.}

%--- Problem 3: Graph of f(x) = 4cos(6x) + 3: period from graph; a, b, d; vertical stretch then translation gives h; describe transformations from y = cos x [10 marks] ---
\item[3.] [Maximum mark: 10]

$f(x) = a\cos(bx) + d$, maximum $(0,\ 7)$, minimum
$\left(\frac{\pi}{6},\ -1\right)$.

\textbf{(a)} period $= 2 \times \dfrac{\pi}{6} = \dfrac{\pi}{3}$
\hfill \textbf{A1}

\textbf{(b)} $a = \dfrac{7 - (-1)}{2} = 4$ \hfill \textbf{A1}

$d = \dfrac{7 + (-1)}{2} = 3$ \hfill \textbf{A1}

$b = \dfrac{2\pi}{\pi/3} = 6$ \hfill \textbf{A1}

\emph{FT $b$ from the candidate's period in (a). Wrong methods, each A0:
$b = \frac{\pi}{3}$ (period used as $b$); $a = 8$ (max $-$ min not
halved); $d = 4$.}

\textbf{(c)} $h(x) = \dfrac{1}{2}\left(4\cos(6x) + 3\right) - 5$
\hfill \textbf{M1}

$= 2\cos(6x) + 1.5 - 5$ \hfill \textbf{A1}

$h(x) = 2\cos(6x) - 3.5$ \hfill \textbf{A1}

\emph{M1 for the stretch applied to the whole of $f$ and the translation
after it. Accept $-\frac{7}{2}$. Wrong methods: stretching only the cosine
term, $2\cos(6x) + 3 - 5 = 2\cos(6x) - 2$, earns M0A0A0; translating
before stretching, $\frac{1}{2}(f(x) - 5) = 2\cos(6x) - 1$, earns M0A0A0.
FT from the candidate's $a$, $b$, $d$.}

\textbf{(d)} Vertical stretch, scale factor $4$ \hfill \textbf{A1}

Horizontal stretch, scale factor $\dfrac{1}{6}$ \hfill \textbf{A1}

Vertical translation $3$ up \hfill \textbf{A1}

\emph{``Describe fully'' needs words (markscheme-spec \S19): one mark per
transformation, with its type, direction and size. Accept ``horizontal
compression by a factor of 6''. Accept ``translation by
$\binom{0}{3}$''. Reject ``move''. A correct chain of equations with no
words earns at most 1 mark. Order: the vertical stretch must come before
the vertical translation; the horizontal stretch may come anywhere.
Deduct 1 mark for a wrong order. Wrong method: horizontal stretch with
scale factor $6$, A0. FT from the candidate's values in (b).}

%--- Problem 4: Tide model D(t) = 2.5sin(0.5t) + 6: max and min depth; period; depth at t = 9; GDC window D >= 7; time of first low tide [11 marks] ---
\item[4.] [Maximum mark: 11]

$D(t) = 2.5\sin(0.5t) + 6$, $t$ hours after midnight.

\textbf{(a)} maximum $= 8.5$ m \hfill \textbf{A1}

minimum $= 3.5$ m \hfill \textbf{A1}

\textbf{(b)} period $= \dfrac{2\pi}{0.5}$ \hfill \textbf{M1}

$= 4\pi = 12.6$ hours \quad $(12.566\ldots)$ \hfill \textbf{A1}

\emph{Accept the exact value $4\pi$. Wrong method: $2\pi \times 0.5 = \pi$
earns M0A0.}

\textbf{(c)} $D(9) = 2.5\sin(0.5 \times 9) + 6$ \hfill \textbf{M1}

$= 3.56$ m \quad $(3.55617\ldots)$ \hfill \textbf{A1}

\emph{Wrong method: degree mode gives $6.20$ (6.19614\ldots), earning
M1A0.}

\textbf{(d)} $2.5\sin(0.5t) + 6 = 7$ \hfill \textbf{M1}

\textit{GDC: graph $y = D(t)$ and $y = 7$ for $0 \le t \le 12$; find both
intersections.}

$t = 0.823$ \ $(0.823033\ldots)$ \ and \ $t = 5.46$ \ $(5.46015\ldots)$
\hfill \textbf{A1}

length $= 5.46015\ldots - 0.823033\ldots = 4.64$ hours
\quad $(4.63711\ldots)$ \hfill \textbf{A1}

\emph{M1 for the equation or inequality $D(t) \ge 7$, or a sketch with the
line $D = 7$. The first A1 needs both values. Accept 4.64 from the
rounded values. Wrong methods: one intersection given as the answer earns
M1A0A0; degree mode gives no intersection in $0 \le t \le 12$, M1A0A0.}

\textbf{(e)} $\sin(0.5t) = -1$, so $0.5t = \dfrac{3\pi}{2}$ \hfill \textbf{M1}

$t = 3\pi = 9.42$ hours after midnight \quad $(9.42477\ldots)$, about
09:25 \hfill \textbf{A1}

\emph{Accept $3\pi$, 9.42 or 09:25. Accept the GDC minimum of the graph
for the M1. Wrong methods: $0.5t = \frac{\pi}{2}$ gives $t = \pi$ (the
first maximum), M0A0; $t = 12.6$ (one period), M0A0.}

\end{enumerate}
