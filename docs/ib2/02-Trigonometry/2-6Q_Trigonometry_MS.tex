\subsubsection*{2.6 Quiz: Trigonometry --- Markscheme}

\textbf{\large Version (a)}

\emph{Total: 34 marks. Accuracy marks need 3 s.f., the exact value, or the
calculator display with an ellipsis. In Problems 1 and 3 the calculator must
be in radian mode: a value worked in degree mode earns the M and
substitution marks if the method is right, and A0 for the answer. Problem
order revised 2026-10-28: the former Problem 2 (two boats) was replaced by
the unit-circle problem.}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: SAS in radians: cosine rule for a side, area, sine rule for an angle in radians [8 marks] ---
\item[1.] [Maximum mark: 8]

\textbf{(a)} $BC^{2} = 6^{2} + 10^{2} - 2(6)(10)\cos 1.1$ \hfill \textbf{M1A1}

$BC^{2} = 81.5684\ldots$

$BC = 9.03\,\text{cm}$ \quad ($9.03152\ldots$) \hfill \textbf{A1}

\emph{Degree mode gives $BC = 4.00$: M1A1A0.}

\textbf{(b)} $\text{Area} = \frac{1}{2}(6)(10)\sin 1.1$ \hfill \textbf{M1}

$\text{Area} = 26.7\,\text{cm}^{2}$ \quad ($26.7362\ldots$) \hfill \textbf{A1}

\emph{Degree mode gives $0.576$: M1A0.}

\textbf{(c)} $\dfrac{\sin A\hat{C}B}{6} = \dfrac{\sin 1.1}{9.03152\ldots}$ \hfill \textbf{M1A1}

$A\hat{C}B = 0.634$ \quad ($0.633617\ldots$) \hfill \textbf{A1}

\emph{M1 for the sine rule, A1 for pairing $AB = 6$ with $A\hat{C}B$. Wrong
method: pairing $10$ with $A\hat{C}B$ gives $1.41$, which is $A\hat{B}C$:
M1A0A0. The answer must be in radians: $36.3^\circ$ earns M1A1A0. Accept
the cosine rule. FT from (a).}

\newpage

%--- Problem 2: Triangular plot (AAS): third angle, sine rule for a side, area; second plot: obtuse angle from area [9 marks] ---
\item[2.] [Maximum mark: 9]

\textbf{(a)} $A\hat{C}B = 180^\circ - 68^\circ - 47^\circ = 65^\circ$ \hfill \textbf{A1}

\textbf{(b)} $\dfrac{AC}{\sin 47^\circ} = \dfrac{40}{\sin 65^\circ}$ \hfill \textbf{M1A1}

$AC = 32.3\,\text{m}$ \quad ($32.2783\ldots$) \hfill \textbf{A1}

\emph{M1 for the sine rule, A1 for pairing $AC$ with $47^\circ$ and $40$
with $65^\circ$. Wrong method: pairing $AC$ with $68^\circ$ gives $40.9$,
which is $BC$: M1A0A0. FT from (a).}

\textbf{(c)} $\text{Area} = \frac{1}{2}(40)(32.2783\ldots)\sin 68^\circ$ \hfill \textbf{M1}

$\text{Area} = 599\,\text{m}^{2}$ \quad ($598.559\ldots$) \hfill \textbf{A1}

\emph{M1 for two sides with the angle between them. Wrong method: using
$\sin 65^\circ$ with $AB$ and $AC$ gives $585$: M0A0. FT from (b).}

\textbf{(d)} $\frac{1}{2}(24)(35)\sin\theta = 320$ \hfill \textbf{M1}

$\sin\theta = \frac{16}{21}$, \quad acute value $49.6^\circ$ \hfill \textbf{(A1)}

$\theta = 180^\circ - 49.6324\ldots^\circ = 130^\circ$ \quad ($130.367\ldots$) \hfill \textbf{A1}

\emph{Giving only the acute value $49.6^\circ$ earns M1A1A0. A1 for $130$
(3 s.f.) or the display with an ellipsis; $130.4$ with no ellipsis earns
A0.}

\newpage

%--- Problem 3: Unit circle: exact coordinates at pi/3, 225 deg in radians and exact coordinates, missing coordinate k, the two solutions of sin(theta) = 0.6 in radians [10 marks] ---
\item[3.] [Maximum mark: 10]

\textbf{(a)} $a = \cos\frac{\pi}{3} = \dfrac{1}{2}$ \hfill \textbf{A1}

$b = \sin\frac{\pi}{3} = \dfrac{\sqrt{3}}{2}$ \hfill \textbf{A1}

\emph{Accept $0.5$ for $a$. $b$ must be exact: $0.866$ earns A0. Wrong
method: swapping sine and cosine ($a = \frac{\sqrt{3}}{2}$,
$b = \frac{1}{2}$): A0A0.}

\textbf{(b)} $225^\circ = 225 \times \dfrac{\pi}{180} = \dfrac{5\pi}{4}$ \hfill \textbf{A1}

\emph{Accept $\frac{225\pi}{180}$ or $1.25\pi$. $3.93$ earns A0 (not a
multiple of $\pi$).}

\textbf{(c)} $c = \cos 225^\circ = -\dfrac{\sqrt{2}}{2}$ \hfill \textbf{A1}

$d = \sin 225^\circ = -\dfrac{\sqrt{2}}{2}$ \hfill \textbf{A1}

\emph{Accept $-\frac{1}{\sqrt{2}}$. A1 for each value with its sign;
$\frac{\sqrt{2}}{2}$ with no minus sign earns A0 for that value. Decimals
($-0.707$) earn A0.}

\textbf{(d)} $k^{2} + 0.6^{2} = 1$ \hfill \textbf{M1}

$k = 0.8$ \hfill \textbf{A1}

\emph{M1 for $x^{2} + y^{2} = 1$ (Pythagoras in the unit circle), or for
$k = \cos\left(\arcsin 0.6\right)$. Accept $\frac{4}{5}$. The negative root
$-0.8$ belongs to $S$: giving $k = \pm 0.8$ as the final answer earns
M1A0. Wrong method: $k = 1 - 0.6 = 0.4$: M0A0.}

\textbf{(e)} $\alpha = \arcsin 0.6 = 0.644$ \quad ($0.643501\ldots$) \hfill \textbf{A1}

$\beta = \pi - \alpha$ \hfill \textbf{M1}

$\beta = 2.50$ \quad ($2.49809\ldots$) \hfill \textbf{A1}

\emph{M1 for $\pi - \alpha$, or for the symmetry of $S$ and $R$ about the
$y$-axis. Answers in degrees ($36.9^\circ$, $143^\circ$) earn A0, M1, A0.
Wrong methods: $\pi + \alpha = 3.79$ or $2\pi - \alpha = 5.64$ (these have
$\sin\theta = -0.6$): M0A0. Accept $2.50$ from the rounded
$\alpha = 0.644$. FT from the candidate's $\alpha$.}

\newpage

%--- Problem 4: Tree: height from tan, line of sight, angle of elevation from a closer point [7 marks] ---
\item[4.] [Maximum mark: 7]

\textbf{(a)} $\tan 32^\circ = \dfrac{FT}{35}$ \hfill \textbf{M1}

$FT = 35\tan 32^\circ = 21.9\,\text{m}$ \quad ($21.8704\ldots$) \hfill \textbf{A1}

\emph{Wrong methods: $35\sin 32^\circ = 18.5$ or
$\frac{35}{\tan 32^\circ} = 56.0$ earn M0A0.}

\textbf{(b)} $AT = \dfrac{35}{\cos 32^\circ}$ \hfill \textbf{M1}

$AT = 41.3\,\text{m}$ \quad ($41.2712\ldots$) \hfill \textbf{A1}

\emph{Accept Pythagoras with $35$ and $FT$, or $\frac{FT}{\sin 32^\circ}$.
Wrong method: $\frac{35}{\sin 32^\circ} = 66.0$ earns M0A0. FT from (a).}

\textbf{(c)} $BF = 35 - 15 = 20$ \hfill \textbf{A1}

$\tan\theta = \dfrac{21.8704\ldots}{20}$ \hfill \textbf{M1}

$\theta = 47.6^\circ$ \quad ($47.5578\ldots$) \hfill \textbf{A1}

\emph{M1 for $\tan\theta = \frac{\text{height}}{\text{their } BF}$. Wrong
method: using $15$ as the horizontal distance gives $55.6^\circ$: A0M1A0.
FT from (a).}

\end{enumerate}

\textbf{\large Version (b)}

\emph{Total: 34 marks. Accuracy marks need 3 s.f., the exact value, or the
calculator display with an ellipsis. In Problem 2 the calculator must be in
radian mode: a value worked in degree mode earns the M and substitution
marks if the method is right, and A0 for the answer.}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Tower seen from two points 50 m apart: show angle ATB, sine rule for BT, height, distance BF [8 marks] ---
\item[1.] [Maximum mark: 8]

\textbf{(a)} $A\hat{B}T = 180^\circ - 40^\circ = 140^\circ$, so
$A\hat{T}B = 180^\circ - 23^\circ - 140^\circ = 17^\circ$ \hfill \textbf{A1}

\hfill \textbf{AG}

\emph{Accept the exterior angle: $40^\circ = 23^\circ + A\hat{T}B$. A1 needs
$140^\circ$ or the exterior-angle reason seen; restating $17^\circ$ earns
A0.}

\textbf{(b)} $\dfrac{BT}{\sin 23^\circ} = \dfrac{50}{\sin 17^\circ}$ \hfill \textbf{M1A1}

$BT = 66.8\,\text{m}$ \quad ($66.8209\ldots$) \hfill \textbf{A1}

\emph{M1 for the sine rule in triangle $ABT$, A1 for pairing $BT$ with
$23^\circ$ and $50$ with $17^\circ$. Wrong method: pairing $BT$ with
$140^\circ$ (or $40^\circ$) gives $109.9$, which is $AT$: M1A0A0.}

\textbf{(c)} $FT = 66.8209\ldots \times \sin 40^\circ$ \hfill \textbf{M1}

$FT = 43.0\,\text{m}$ \quad ($42.9516\ldots$) \hfill \textbf{A1}

\emph{Accept $AT\sin 23^\circ$ with $AT = 109.926\ldots$. Accept $42.9$
from the rounded $BT = 66.8$. Wrong method: $BT\cos 40^\circ = 51.2$ (the
distance $BF$): M0A0. FT from (b).}

\textbf{(d)} $BF = 66.8209\ldots \times \cos 40^\circ$ \hfill \textbf{M1}

$BF = 51.2\,\text{m}$ \quad ($51.1878\ldots$) \hfill \textbf{A1}

\emph{Accept $\frac{FT}{\tan 40^\circ}$, Pythagoras with $BT$ and $FT$, or
$AT\cos 23^\circ - 50$. Wrong method: $BT\sin 40^\circ = 43.0$ (the height):
M0A0. FT from (b) or (c).}

\newpage

%--- Problem 2: Ambiguous case in radians: two values of angle PRQ, then the obtuse case: third angle, PR, area [8 marks] ---
\item[2.] [Maximum mark: 8]

\textbf{(a)} $\dfrac{\sin P\hat{R}Q}{12} = \dfrac{\sin 0.7}{9}$ \hfill \textbf{M1}

$\sin P\hat{R}Q = 0.858956\ldots$

$P\hat{R}Q = 1.03$ \quad ($1.03322\ldots$) \hfill \textbf{A1}

$P\hat{R}Q = \pi - 1.03322\ldots = 2.11$ \quad ($2.10836\ldots$) \hfill \textbf{A1}

\emph{Both values are possible because $0.7 + 2.10836\ldots < \pi$. Wrong
methods: giving only $1.03$ earns M1A1A0; degree mode ($\sin P\hat{R}Q =
0.0163$) earns M1A0A0; answers in degrees ($59.2^\circ$, $121^\circ$) earn
M1A0A0.}

\textbf{(b)} $P\hat{Q}R = \pi - 0.7 - 2.10836\ldots = 0.333$ \quad ($0.333229\ldots$) \hfill \textbf{A1}

\emph{Accept $0.332$ from the rounded $P\hat{R}Q = 2.11$. Wrong method:
using the acute value gives $1.41$: A0. FT from (a).}

\textbf{(c)} $\dfrac{PR}{\sin 0.333229\ldots} = \dfrac{9}{\sin 0.7}$ \hfill \textbf{M1}

$PR = 4.57\,\text{cm}$ \quad ($4.56967\ldots$) \hfill \textbf{A1}

\emph{Accept the cosine rule $9^{2} = 12^{2} + PR^{2} - 2(12)(PR)\cos 0.7$:
it gives $PR = 4.57$ or $13.8$, and the obtuse case is the smaller root;
giving $13.8$ (the acute case) earns M1A0. Accept $4.55$ from the rounded
$P\hat{R}Q = 2.11$. FT from (b).}

\textbf{(d)} $\text{Area} = \frac{1}{2}(12)(9)\sin 0.333229\ldots$ \hfill \textbf{M1}

$\text{Area} = 17.7\,\text{cm}^{2}$ \quad ($17.6631\ldots$) \hfill \textbf{A1}

\emph{Accept $\frac{1}{2}(12)(PR)\sin 0.7$. Accept $17.6$ from the rounded
$P\hat{R}Q = 2.11$. Wrong method: $\frac{1}{2}(12)(9)\sin 0.7 = 34.8$ (0.7
is not between these sides): M0A0. FT from (b) or (c).}

\emph{Feedback: in this problem rounding an angle early moves the later
answers. Store the full values in the calculator.}

%--- Problem 3: Quadrilateral ABCD split by diagonal AC: cosine rule for AC, SSS cosine rule for angle ADC, area [9 marks] ---
\item[3.] [Maximum mark: 9]

\textbf{(a)} $AC^{2} = 8^{2} + 5^{2} - 2(8)(5)\cos 125^\circ$ \hfill \textbf{M1A1}

$AC^{2} = 134.886\ldots$

$AC = 11.6\,\text{cm}$ \quad ($11.6140\ldots$) \hfill \textbf{A1}

\emph{M1 for the cosine rule, A1 for the correct substitution. Wrong method:
a sign error ($89 - 45.9$) gives $6.57$: M1A0A0.}

\textbf{(b)} $\cos A\hat{D}C = \dfrac{7^{2} + 12^{2} - 11.6140\ldots^{2}}{2(7)(12)}$ \hfill \textbf{M1A1}

$\cos A\hat{D}C = 0.345915\ldots$

$A\hat{D}C = 69.8^\circ$ \quad ($69.7622\ldots$) \hfill \textbf{A1}

\emph{M1 for the cosine rule for an angle, A1 for $AC$ as the side opposite
$A\hat{D}C$. Accept $69.6^\circ$ from the rounded $AC = 11.6$. Wrong method:
using $AD = 12$ as the opposite side gives $75.8^\circ$, which is
$A\hat{C}D$: M1A0A0. FT from (a).}

\textbf{(c)} $\text{Area } ABC = \frac{1}{2}(8)(5)\sin 125^\circ = 16.3830\ldots$

$\text{Area } ACD = \frac{1}{2}(7)(12)\sin 69.7622\ldots^\circ = 39.4071\ldots$ \hfill \textbf{M1}

$\text{Area } ABCD = 16.3830\ldots + 39.4071\ldots$ \hfill \textbf{M1}

$\text{Area } ABCD = 55.8\,\text{cm}^{2}$ \quad ($55.7901\ldots$) \hfill \textbf{A1}

\emph{First M1 for a correct area expression for either triangle (two sides
and the angle between them). Second M1 for adding the two triangles. Wrong
method: using $125^\circ$ in triangle $ACD$ as well gives $50.8$: M1M1A0.
FT from (b).}

\newpage

%--- Problem 4: Grouped data with two unknown frequencies; estimated sd; linear transformation of mean and sd [9 marks] ---
\item[4.] [Maximum mark: 9]

\textbf{(a)} $6 + p + 12 + q + 3 = 40$, so $p + q = 19$ \hfill \textbf{A1}

$\dfrac{15(6) + 25p + 35(12) + 45q + 55(3)}{40} = 31.25$ \hfill \textbf{M1}

$675 + 25p + 45q = 1250$, so $5p + 9q = 115$

$p = 14$ \hfill \textbf{A1}

$q = 5$ \hfill \textbf{A1}

\emph{M1 for the mean equation using the class midpoints $15, 25, \ldots,
55$. Accept solving the system on the GDC. Wrong method: lower class
boundaries ($10, 20, \ldots$) give $p = 4$, $q = 15$: A1M0A0A0.}

\textbf{(b)} $\sigma = 11.1$ minutes \quad ($11.1102\ldots$) \hfill \textbf{M1A1}

\emph{M1 for the midpoints with their frequencies entered in the GDC (may
be implied by a correct answer). Wrong method: the sample value
$s_{x} = 11.3$ earns M1A0. FT from (a).}

\textbf{(c)} new time $= 60(t + 2) = 60t + 120$ seconds \hfill \textbf{(M1)}

new mean $= 60(31.25 + 2) = 1995\,\text{s}$ \hfill \textbf{A1}

new standard deviation $= 60 \times 11.1102\ldots = 667\,\text{s}$ \quad ($666.614\ldots$) \hfill \textbf{A1}

\emph{The penalty moves the mean but not the spread; the conversion to
seconds multiplies both by $60$. Accept $666$ from the rounded $\sigma =
11.1$. Wrong methods: adding $120$ to the standard deviation ($787$), or
adding the $2$ minutes after converting (mean $1877$): A0 for that value.
FT from (b).}

\end{enumerate}
