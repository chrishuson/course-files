\subsubsection*{2.2 Classwork: The Cosine Rule --- Markscheme}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Cosine rule on three triangles: side from SAS, largest angle from SSS (obtuse), classify, exact side with 120 deg [7 marks] ---
\item[1.] [Maximum mark: 7]

\textbf{(a)} $x^{2} = 8^{2} + 11^{2} - 2(8)(11)\cos 40^\circ$ \hfill \textbf{M1}

$x = 7.08$ \quad ($7.08351\ldots$) \hfill \textbf{A1}

\emph{M1 for a correct substitution into the cosine rule. A value from a
calculator in radian mode ($x = 17.4$) earns M1A0.}

\textbf{(b)} The largest angle is opposite the longest side, $9$.

$\cos\theta = \frac{5^{2} + 7^{2} - 9^{2}}{2(5)(7)}$ \hfill \textbf{M1}

$\cos\theta = -\frac{7}{70} = -0.1$

$\theta = 95.7^\circ$ \quad ($95.7391\ldots$) \hfill \textbf{A1}

\emph{M1 for a correct substitution with $9^{2}$ as the subtracted term. An
angle found opposite the side $5$ or $7$ earns M0A0.}

\textbf{(c)} obtuse, since $\theta > 90^\circ$ \hfill \textbf{A1}

\emph{Accept ``obtuse, since $\cos\theta < 0$''. FT from (b).}

\textbf{(d)} $y^{2} = 6^{2} + 10^{2} - 2(6)(10)\cos 120^\circ$ \hfill \textbf{M1}

$y^{2} = 36 + 100 + 60$

$y^{2} = 196$

$y = 14$ \hfill \textbf{A1}

\emph{The common error is a sign slip with $\cos 120^\circ = -\frac{1}{2}$,
giving $y^{2} = 76$ and $y = 8.72$: M1A0.}

%--- Problem 2: Hikers 6 km and 9 km at 35 deg (distance apart); quadrilateral ABCD: diagonal AC from SAS, angle ADC from SSS [6 marks] ---
\item[2.] [Maximum mark: 6]

\textbf{(a)} $d^{2} = 6^{2} + 9^{2} - 2(6)(9)\cos 35^\circ$ \hfill \textbf{M1}

$d = 5.34\,\text{km}$ \quad ($5.34149\ldots$) \hfill \textbf{A1}

\emph{A diagram is not required.}

\textbf{(b)} $AC^{2} = 5^{2} + 8^{2} - 2(5)(8)\cos 60^\circ$ \hfill \textbf{M1}

$AC^{2} = 25 + 64 - 40$

$AC^{2} = 49$

$AC = 7\,\text{cm}$ \hfill \textbf{A1}

\textbf{(c)} $\cos \angle ADC = \frac{4^{2} + 6^{2} - 7^{2}}{2(4)(6)}$ \hfill \textbf{M1}

$\cos \angle ADC = \frac{3}{48}$

$\angle ADC = 86.4^\circ$ \quad ($86.4166\ldots$) \hfill \textbf{A1}

\emph{M1 for a correct substitution with $AC^{2}$ as the subtracted term.
FT from (b).}

\end{enumerate}
