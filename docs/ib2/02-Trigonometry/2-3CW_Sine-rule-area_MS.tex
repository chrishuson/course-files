\subsubsection*{2.3 Classwork: The Sine Rule and the Area of a Triangle --- Markscheme}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Sine rule for a side (AAS), area after third angle, sine rule for an acute angle (SSA), area SAS exact, angle from area [11 marks] ---
\item[1.] [Maximum mark: 11]

\textbf{(a)} $\frac{AC}{\sin 65^\circ} = \frac{8}{\sin 40^\circ}$ \hfill \textbf{M1}

$AC = \frac{8\sin 65^\circ}{\sin 40^\circ}$

$AC = 11.3$ \quad ($11.2797\ldots$) \hfill \textbf{A1}

\emph{M1 for a correct sine rule equation pairing each side with its
opposite angle.}

\textbf{(b)} $C = 180^\circ - 40^\circ - 65^\circ = 75^\circ$ \hfill \textbf{A1}

$\text{Area} = \frac{1}{2}(8)(11.2797\ldots)\sin 75^\circ$ \hfill \textbf{M1}

$\text{Area} = 43.6$ \quad ($43.5814\ldots$) \hfill \textbf{A1}

\emph{M1 for $\frac{1}{2}ab\sin C$ with two sides and the angle between
them. Accept $43.7$ from the rounded $AC = 11.3$. FT from (a).}

\textbf{(c)} $\frac{\sin R}{10} = \frac{\sin 35^\circ}{8}$ \hfill \textbf{M1}

$\sin R = \frac{10\sin 35^\circ}{8} = 0.716970\ldots$

$R = 45.8^\circ$ \quad ($45.8049\ldots$) \hfill \textbf{A1}

\emph{The obtuse value $134.2^\circ$ is excluded by the statement that $R$
is acute.}

\textbf{(d)} $\text{Area} = \frac{1}{2}(6)(9)\sin 30^\circ$ \hfill \textbf{M1}

$\text{Area} = \frac{27}{2}$ \hfill \textbf{A1}

\emph{Accept $13.5$.}

\textbf{(e)} $\frac{1}{2}(5)(10)\sin\theta = 20$ \hfill \textbf{M1}

$\sin\theta = 0.8$

$\theta = 53.1^\circ$ \quad ($53.1301\ldots$) \hfill \textbf{A1}

\newpage

%--- Problem 2: River width: third angle, sine rule for AT, right-angle trig for width, area of ABT; garden angle from area [9 marks] ---
\item[2.] [Maximum mark: 9]

\textbf{(a)} $\angle ATB = 180^\circ - 52^\circ - 71^\circ = 57^\circ$ \hfill \textbf{A1}

\textbf{(b)} $\frac{AT}{\sin 71^\circ} = \frac{200}{\sin 57^\circ}$ \hfill \textbf{M1}

$AT = 225\,\text{m}$ \quad ($225.480\ldots$) \hfill \textbf{A1}

\emph{FT from (a).}

\textbf{(c)} width $= AT \sin 52^\circ$ \hfill \textbf{M1}

width $= 178\,\text{m}$ \quad ($177.680\ldots$) \hfill \textbf{A1}

\emph{M1 for right-angled trigonometry in the triangle formed by $T$, $A$
and the foot of the perpendicular from $T$. Accept $BT \sin 71^\circ$ with
$BT = 187.92\ldots$ from the sine rule. Accept $177$ from the rounded
$AT = 225$. FT from (b).}

\textbf{(d)} $\text{Area} = \frac{1}{2}(200)(225.480\ldots)\sin 52^\circ$ \hfill \textbf{M1}

$\text{Area} = 17\,800\,\text{m}^{2}$ \quad ($17\,768.0\ldots$) \hfill \textbf{A1}

\emph{Accept $\frac{1}{2}(200)(\text{width})$. Accept $17\,700$ from the
rounded $AT = 225$. FT from (b) or (c).}

\textbf{(e)} $\frac{1}{2}(8)(10)\sin\theta = 30$ \hfill \textbf{M1}

$\sin\theta = 0.75$

$\theta = 48.6^\circ$ \quad ($48.5903\ldots$) \hfill \textbf{A1}

\end{enumerate}
