\subsubsection*{2.4 Pre-Quiz: Sine Rule, Cosine Rule and the Area of a Triangle --- Markscheme}

\emph{Total: 40 marks. Accuracy marks need 3 s.f., the exact value, or the
calculator display with an ellipsis. In
Problems 2, 4 and 6 the calculator must be in radian mode: a value worked
in degree mode earns the M mark if the method is right, and A0.}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Obtuse angle given (SSA): sine rule for an angle, third angle + sine rule for a side, area [7 marks] ---
\item[1.] [Maximum mark: 7]

\textbf{(a)} $\dfrac{\sin B\hat{A}C}{15} = \dfrac{\sin 108^\circ}{21}$ \hfill \textbf{M1}

$\sin B\hat{A}C = \dfrac{15\sin 108^\circ}{21} = 0.679326\ldots$

$B\hat{A}C = 42.8^\circ$ \quad ($42.7910\ldots$) \hfill \textbf{A1}

\emph{M1 for a correct sine rule equation pairing each side with its
opposite angle, in either orientation. The obtuse value $137.2^\circ$ is
impossible because the triangle already has an obtuse angle at $B$.}

\textbf{(b)} $A\hat{C}B = 180^\circ - 108^\circ - 42.7910\ldots^\circ = 29.2^\circ$ \quad ($29.2089\ldots$) \hfill \textbf{A1}

$\dfrac{AB}{\sin 29.2089\ldots^\circ} = \dfrac{21}{\sin 108^\circ}$ \hfill \textbf{M1}

$AB = 10.8\,\text{cm}$ \quad ($10.7753\ldots$) \hfill \textbf{A1}

\emph{Accept the cosine rule:
$21^{2} = AB^{2} + 15^{2} - 2(15)(AB)\cos 108^\circ$, solved on the GDC
(the A1 for $A\hat{C}B$ is then implied by a correct final answer). FT
from (a).}

\textbf{(c)} $\text{Area} = \frac{1}{2}(15)(21)\sin 29.2089\ldots^\circ$ \hfill \textbf{M1}

$\text{Area} = 76.9\,\text{cm}^{2}$ \quad ($76.8594\ldots$) \hfill \textbf{A1}

\emph{Accept any two sides with the angle between them, e.g.
$\frac{1}{2}(15)(AB)\sin 108^\circ$. Accept $76.8$ from the rounded
$A\hat{C}B = 29.2^\circ$. FT from (a) or (b).}

\emph{Feedback: carry the full calculator value of each angle forward;
round only the final answer.}

\newpage

%--- Problem 2: SAS in radians: cosine rule for a side, area, sine rule for an angle in radians [6 marks] ---
\item[2.] [Maximum mark: 6]

\textbf{(a)} $QR^{2} = 9^{2} + 7^{2} - 2(9)(7)\cos 1.3$ \hfill \textbf{M1}

$QR^{2} = 96.2951\ldots$

$QR = 9.81\,\text{cm}$ \quad ($9.81300\ldots$) \hfill \textbf{A1}

\emph{Degree mode gives $QR = 2.01$: M1A0.}

\textbf{(b)} $\text{Area} = \frac{1}{2}(9)(7)\sin 1.3$ \hfill \textbf{M1}

$\text{Area} = 30.4\,\text{cm}^{2}$ \quad ($30.3520\ldots$) \hfill \textbf{A1}

\emph{Degree mode gives $0.715$: M1A0.}

\textbf{(c)} $\dfrac{\sin P\hat{Q}R}{7} = \dfrac{\sin 1.3}{9.81300\ldots}$ \hfill \textbf{M1}

$\sin P\hat{Q}R = 0.687343\ldots$

$P\hat{Q}R = 0.758$ \quad ($0.757825\ldots$) \hfill \textbf{A1}

\emph{Accept the cosine rule,
$\cos P\hat{Q}R = \frac{9^{2} + QR^{2} - 7^{2}}{2(9)(QR)}$. The answer
must be in radians: $43.4^\circ$ earns M1A0. $P\hat{Q}R$ is acute because
it is opposite the shortest side. FT from (a).}

\newpage

%--- Problem 3: Triangular field (SSS): cosine rule for an angle, area, perpendicular distance from area [6 marks] ---
\item[3.] [Maximum mark: 6]

\textbf{(a)} $\cos A\hat{B}C = \dfrac{120^{2} + 85^{2} - 140^{2}}{2(120)(85)}$ \hfill \textbf{M1}

$\cos A\hat{B}C = \dfrac{2025}{20\,400} = 0.0992647\ldots$

$A\hat{B}C = 84.3^\circ$ \quad ($84.3031\ldots$) \hfill \textbf{A1}

\emph{M1 for the cosine rule with $AC = 140$ as the side opposite
$A\hat{B}C$.}

\textbf{(b)} $\text{Area} = \frac{1}{2}(120)(85)\sin 84.3031\ldots^\circ$ \hfill \textbf{M1}

$\text{Area} = 5070\,\text{m}^{2}$ \quad ($5074.81\ldots$) \hfill \textbf{A1}

\emph{Accept $5.07 \times 10^{3}$. Accept any two sides with the angle
between them, e.g. $\frac{1}{2}(120)(140)\sin B\hat{A}C$ with
$B\hat{A}C = 37.2^\circ$. FT from (a).}

\textbf{(c)} $\frac{1}{2}(140)h = 5074.81\ldots$ \hfill \textbf{M1}

$h = 72.5\,\text{m}$ \quad ($72.4973\ldots$) \hfill \textbf{A1}

\emph{M1 for recognizing that the shortest route is perpendicular to $AC$,
by any method: the area with base $AC$, or $h = 120\sin B\hat{A}C$, or
$h = 85\sin B\hat{C}A$. Accept $72.4$ from the rounded area $5070$. FT from
(b).}

\emph{Feedback: carry the full area into (c); round only the final
answer.}

\newpage

%--- Problem 4: Ambiguous case in radians: two values of an angle, third angle, area [6 marks] ---
\item[4.] [Maximum mark: 6]

\textbf{(a)} $\dfrac{\sin A\hat{C}B}{10} = \dfrac{\sin 0.6}{7}$ \hfill \textbf{M1}

$\sin A\hat{C}B = \dfrac{10\sin 0.6}{7} = 0.806632\ldots$

$A\hat{C}B = 0.938$ \quad ($0.938431\ldots$) \hfill \textbf{A1}

$A\hat{C}B = \pi - 0.938431\ldots = 2.20$ \quad ($2.20316\ldots$) \hfill \textbf{A1}

\emph{Both values are possible because $0.6 + 2.20316\ldots < \pi$. Degree
mode gives $\sin A\hat{C}B = 0.0150$: M1A0A0. Answers in degrees
($53.8^\circ$, $126^\circ$) earn M1A0A0.}

\textbf{(b)} $A\hat{B}C = \pi - 0.6 - 2.20316\ldots$

$A\hat{B}C = 0.338$ \quad ($0.338431\ldots$) \hfill \textbf{A1}

\emph{A bare correct answer earns A1. Accept $0.342$ from the rounded
$A\hat{C}B = 2.20$. FT from (a).}

\textbf{(c)} $\text{Area} = \frac{1}{2}(10)(7)\sin 0.338431\ldots$ \hfill \textbf{M1}

$\text{Area} = 11.6\,\text{cm}^{2}$ \quad ($11.6202\ldots$) \hfill \textbf{A1}

\emph{Accept any two sides with the angle between them, e.g.
$\frac{1}{2}(10)(AC)\sin 0.6$ with $AC = 4.12$ from the sine rule. Accept
$11.7$ from the rounded $A\hat{C}B = 2.20$. FT from (b).}

\emph{Feedback: here rounding $A\hat{C}B$ to $2.20$ moves the area from
$11.6$ to $11.7$. Store the full value in the calculator and use it.}

\newpage

%--- Problem 5: Quadrilateral split by a diagonal: cosine rule for BD, sine rule for an angle, area of quadrilateral [8 marks] ---
\item[5.] [Maximum mark: 8]

\textbf{(a)} $BD^{2} = 6^{2} + 9^{2} - 2(6)(9)\cos 75^\circ$ \hfill \textbf{M1}

$BD^{2} = 89.0475\ldots$

$BD = 9.44\,\text{cm}$ \quad ($9.43650\ldots$) \hfill \textbf{A1}

\textbf{(b)} $\dfrac{\sin B\hat{D}C}{8} = \dfrac{\sin 62^\circ}{9.43650\ldots}$ \hfill \textbf{M1}

$\sin B\hat{D}C = 0.748538\ldots$

$B\hat{D}C = 48.5^\circ$ \quad ($48.4639\ldots$) \hfill \textbf{A1}

\emph{The obtuse value $131.5^\circ$ is impossible:
$131.5^\circ + 62^\circ > 180^\circ$. Accept $48.4^\circ$ from the rounded
$BD = 9.44$. FT from (a).}

\textbf{(c)} $C\hat{B}D = 180^\circ - 62^\circ - 48.4639\ldots^\circ = 69.5360\ldots^\circ$ \hfill \textbf{A1}

$\text{Area } ABD = \frac{1}{2}(6)(9)\sin 75^\circ = 26.0799\ldots$

$\text{Area } BCD = \frac{1}{2}(8)(9.43650\ldots)\sin 69.5360\ldots^\circ = 35.3639\ldots$ \hfill \textbf{M1}

$\text{Area } ABCD = 26.0799\ldots + 35.3639\ldots$ \hfill \textbf{M1}

$\text{Area } ABCD = 61.4\,\text{cm}^{2}$ \quad ($61.4439\ldots$) \hfill \textbf{A1}

\emph{A1 for $C\hat{B}D$ (may be seen in the area expression). First M1
for a correct area expression for triangle $BCD$ with two sides and the
angle between them, by any route, e.g. $\frac{1}{2}(BD)(CD)\sin B\hat{D}C$
with $CD = 10.0$ from the sine rule. Second M1 for adding the areas of
the two triangles. Accept $61.5$ from the rounded $BD = 9.44$ and
$B\hat{D}C = 48.4^\circ$. FT from (a) and (b).}

\emph{Feedback: carry $BD$ and $B\hat{D}C$ forward at full precision.}

\newpage

%--- Problem 6: Area equation in x (radians): solve for x, cosine rule for BC, sine rule for an angle in radians [7 marks] ---
\item[6.] [Maximum mark: 7]

\textbf{(a)} Attempt to use the area formula with $x$, $x + 4$ and $2.1$ \hfill \textbf{(M1)}

$\frac{1}{2}x(x + 4)\sin 2.1 = 30$ \hfill \textbf{A1}

$x = 6.57$ \quad ($6.57368\ldots$) \hfill \textbf{A1}

\emph{Accept the equivalent quadratic $x^{2} + 4x - 69.5080\ldots = 0$.
The root $x = -10.6$ is rejected because a length is positive; giving both
roots as the final answer earns A0. Degree mode gives $x = 38.5$: A0.}

\textbf{(b)} $BC^{2} = 6.57368\ldots^{2} + 10.5736\ldots^{2} - 2(6.57368\ldots)(10.5736\ldots)\cos 2.1$ \hfill \textbf{M1}

$BC^{2} = 225.197\ldots$

$BC = 15.0\,\text{cm}$ \quad ($15.0065\ldots$) \hfill \textbf{A1}

\emph{FT from (a).}

\textbf{(c)} $\dfrac{\sin A\hat{B}C}{10.5736\ldots} = \dfrac{\sin 2.1}{15.0065\ldots}$ \hfill \textbf{M1}

$\sin A\hat{B}C = 0.608219\ldots$

$A\hat{B}C = 0.654$ \quad ($0.653815\ldots$) \hfill \textbf{A1}

\emph{M1 for a sine rule or cosine rule equation with $AC = x + 4$ as the
side opposite $A\hat{B}C$. $A\hat{B}C$ is acute because $B\hat{A}C$ is
obtuse. The answer must be in radians: $37.5^\circ$ earns M1A0. FT from
(a) and (b).}

\end{enumerate}
