\subsubsection*{2.6 Early Finishers: Trigonometry and Statistics --- Markscheme}

\textbf{\large Version (a)}

\emph{Extension set, scored separately from the quiz. Total: 17 marks. Both
problems are copies of quiz version (b) problems 2 and 4, with the same
markscheme. Accuracy marks need 3 s.f., the exact value, or the calculator
display with an ellipsis. Problem 1 needs radian mode.}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 2: Ambiguous case in radians: two values of angle PRQ, then the obtuse case: third angle, PR, area [8 marks] ---
\item[1.] [Maximum mark: 8]

\textbf{(a)} $\dfrac{\sin P\hat{R}Q}{12} = \dfrac{\sin 0.7}{9}$ \hfill \textbf{M1}

$\sin P\hat{R}Q = 0.858956\ldots$

$P\hat{R}Q = 1.03$ \quad ($1.03322\ldots$) \hfill \textbf{A1}

$P\hat{R}Q = \pi - 1.03322\ldots = 2.11$ \quad ($2.10836\ldots$) \hfill \textbf{A1}

\emph{Both values are possible because $0.7 + 2.10836\ldots < \pi$. Wrong
methods: giving only $1.03$ earns M1A1A0; degree mode ($\sin P\hat{R}Q =
0.0163$) earns M1A0A0; answers in degrees ($59.2^\circ$, $121^\circ$) earn
M1A0A0.}

\textbf{(b)} $P\hat{Q}R = \pi - 0.7 - 2.10836\ldots = 0.333$ \quad ($0.333229\ldots$) \hfill \textbf{A1}

\emph{Accept $0.332$ from the rounded $P\hat{R}Q = 2.11$. Wrong method:
using the acute value gives $1.41$: A0. FT from (a).}

\textbf{(c)} $\dfrac{PR}{\sin 0.333229\ldots} = \dfrac{9}{\sin 0.7}$ \hfill \textbf{M1}

$PR = 4.57\,\text{cm}$ \quad ($4.56967\ldots$) \hfill \textbf{A1}

\emph{Accept the cosine rule $9^{2} = 12^{2} + PR^{2} - 2(12)(PR)\cos 0.7$:
it gives $PR = 4.57$ or $13.8$, and the obtuse case is the smaller root;
giving $13.8$ (the acute case) earns M1A0. Accept $4.55$ from the rounded
$P\hat{R}Q = 2.11$. FT from (b).}

\textbf{(d)} $\text{Area} = \frac{1}{2}(12)(9)\sin 0.333229\ldots$ \hfill \textbf{M1}

$\text{Area} = 17.7\,\text{cm}^{2}$ \quad ($17.6631\ldots$) \hfill \textbf{A1}

\emph{Accept $\frac{1}{2}(12)(PR)\sin 0.7$. Accept $17.6$ from the rounded
$P\hat{R}Q = 2.11$. Wrong method: $\frac{1}{2}(12)(9)\sin 0.7 = 34.8$ (0.7
is not between these sides): M0A0. FT from (b) or (c).}

\emph{Feedback: in this problem rounding an angle early moves the later
answers. Store the full values in the calculator.}

\newpage

%--- Problem 4: Grouped data with two unknown frequencies; estimated sd; linear transformation of mean and sd [9 marks] ---
\item[2.] [Maximum mark: 9]

\textbf{(a)} $6 + p + 12 + q + 3 = 40$, so $p + q = 19$ \hfill \textbf{A1}

$\dfrac{15(6) + 25p + 35(12) + 45q + 55(3)}{40} = 31.25$ \hfill \textbf{M1}

$675 + 25p + 45q = 1250$, so $5p + 9q = 115$

$p = 14$ \hfill \textbf{A1}

$q = 5$ \hfill \textbf{A1}

\emph{M1 for the mean equation using the class midpoints $15, 25, \ldots,
55$. Accept solving the system on the GDC. Wrong method: lower class
boundaries ($10, 20, \ldots$) give $p = 4$, $q = 15$: A1M0A0A0.}

\textbf{(b)} $\sigma = 11.1$ minutes \quad ($11.1102\ldots$) \hfill \textbf{M1A1}

\emph{M1 for the midpoints with their frequencies entered in the GDC (may
be implied by a correct answer). Wrong method: the sample value
$s_{x} = 11.3$ earns M1A0. FT from (a).}

\textbf{(c)} new time $= 60(t + 2) = 60t + 120$ seconds \hfill \textbf{(M1)}

new mean $= 60(31.25 + 2) = 1995\,\text{s}$ \hfill \textbf{A1}

new standard deviation $= 60 \times 11.1102\ldots = 667\,\text{s}$ \quad ($666.614\ldots$) \hfill \textbf{A1}

\emph{The penalty moves the mean but not the spread; the conversion to
seconds multiplies both by $60$. Accept $666$ from the rounded $\sigma =
11.1$. Wrong methods: adding $120$ to the standard deviation ($787$), or
adding the $2$ minutes after converting (mean $1877$): A0 for that value.
FT from (b).}

\end{enumerate}

\textbf{\large Version (b)}

\emph{Extension set, scored separately from the quiz. Total: 21 marks.
Accuracy marks need 3 s.f., the exact value, or the calculator display with
an ellipsis. Problem 1 needs radian mode: a value worked in degree mode
earns the M and substitution marks if the method is right, and A0 for the
answer.}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Kite in radians: diagonal BD, angle BCD (obtuse), area, angle ABC [11 marks] ---
\item[1.] [Maximum mark: 11]

\textbf{(a)} $BD^{2} = 10^{2} + 10^{2} - 2(10)(10)\cos 1.4$ \hfill \textbf{M1A1}

$BD^{2} = 166.006\ldots$

$BD = 12.9\,\text{cm}$ \quad ($12.8843\ldots$) \hfill \textbf{A1}

\emph{M1 for the cosine rule in triangle $ABD$, A1 for the correct
substitution. Degree mode gives $0.244$: M1A1A0.}

\textbf{(b)} $\cos B\hat{C}D = \dfrac{7^{2} + 7^{2} - 12.8843\ldots^{2}}{2(7)(7)}$ \hfill \textbf{M1A1}

$\cos B\hat{C}D = -0.693944\ldots$

$B\hat{C}D = 2.34$ \quad ($2.33774\ldots$) \hfill \textbf{A1}

\emph{M1 for the cosine rule for an angle in triangle $BCD$, A1 for $BD$ as
the side opposite $B\hat{C}D$. The answer must be in radians: $134^\circ$
earns M1A1A0. Wrong method: a sign error gives $0.804$: M1A0A0. Accept
$2.34$ from the rounded $BD = 12.9$. FT from (a).}

\textbf{(c)} $\text{Area } ABD = \frac{1}{2}(10)(10)\sin 1.4 = 49.2724\ldots$

$\text{Area } BCD = \frac{1}{2}(7)(7)\sin 2.33774\ldots = 17.6407\ldots$ \hfill \textbf{M1}

$\text{Area} = 49.2724\ldots + 17.6407\ldots$ \hfill \textbf{M1}

$\text{Area} = 66.9\,\text{cm}^{2}$ \quad ($66.9131\ldots$) \hfill \textbf{A1}

\emph{First M1 for a correct area expression for either triangle. Second M1
for adding the two triangles. Accept other correct methods: two copies of
triangle $ABC$, $2 \times \frac{1}{2}(10)(7)\sin A\hat{B}C$, or half the
product of the diagonals. Accept any value consistent with the
candidate's own correctly rounded intermediates, e.g.\ $66.8$ from
$BD = 12.9$. Wrong method: triangle $ABD$ alone, $49.3$: M1M0A0. FT from
(b).}

\textbf{(d)} $2A\hat{B}C = 2\pi - 1.4 - 2.33774\ldots$ \hfill \textbf{M1}

$A\hat{B}C = 1.27$ \quad ($1.27271\ldots$) \hfill \textbf{A1}

\emph{M1 for the angle sum $2\pi$ with $A\hat{B}C = A\hat{D}C$ by symmetry,
or for any other valid method (e.g.\ $A\hat{B}D + D\hat{B}C$). Wrong
method: $A\hat{B}D = \frac{\pi - 1.4}{2} = 0.871$ alone: M0A0. $72.9^\circ$
earns M1A0. FT from (b).}

\newpage

%--- Problem 2: Bivariate data: k from the mean; r and the y-on-x line; x-on-y line to estimate x; intersection at the mean point [10 marks] ---
\item[2.] [Maximum mark: 10]

\textbf{(a)} $\dfrac{12 + 19 + 14 + k + 27 + 26}{6} = 20$ \hfill \textbf{M1}

$98 + k = 120$, \quad $k = 22$ \hfill \textbf{A1}

\emph{Wrong method: dividing by $5$ gives $k = 2$: M0A0.}

\textbf{(b)} $r = 0.896$ \quad ($0.895547\ldots$) \hfill \textbf{A1}

$y = 1.90x + 8.57$ \quad ($1.90476\ldots$, $8.57142\ldots$) \hfill \textbf{A1A1}

\emph{A1 for $a$, A1 for $b$. Accept $a = \frac{40}{21}$, $b = \frac{60}{7}$.
The answer must be an equation; award A1A0 if $a$ and $b$ are given
without the equation. FT from (a).}

\textbf{(c)} use the regression line of $x$ on $y$ \hfill \textbf{M1}

$x = 0.421y - 2.42$ \quad ($0.421052\ldots$, $-2.42105\ldots$) \hfill \textbf{A1}

$x = 0.421052\ldots(24) - 2.42105\ldots = 7.68$ hours \quad ($7.68421\ldots$) \hfill \textbf{A1}

\emph{M1 for choosing the $x$-on-$y$ line to estimate $x$ from $y$. Accept
$7.68$ from the rounded coefficients. Wrong method: solving the $y$-on-$x$
line for $x$ gives $8.10$: M0A0A0. FT from (a).}

\textbf{(d)} $(\bar{x},\ \bar{y}) = (6,\ 20)$ \hfill \textbf{A1A1}

\emph{A1 for each coordinate. Both regression lines pass through the mean
point $(\bar{x},\ \bar{y})$; $\bar{y} = 20$ is given in the question.}

\end{enumerate}
