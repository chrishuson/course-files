\subsubsection*{7.8 Quiz: Logarithms (no calculator)}

\begin{enumerate}[itemsep=0.1cm]

%--- Problem 1: Express in form ln k [5 marks] ---
\item[1.] [Maximum mark: 5]

Write each of the following expressions in the form $\ln k$, where
$k \in \mathbb{Z}^{+}$.

\begin{enumerate}[itemsep=0.1cm]
  \item $\ln 4 + \ln 7$ \hfill [1]
  \item $2\ln 5$ \hfill [2]
  \item $-\ln\dfrac{1}{6}$ \hfill [2]
\end{enumerate}

\medskip

%--- Problem 2: log_10 expressions in p, q [5 marks] ---
\item[2.] [Maximum mark: 5]

Let $\log_{10} 3 = p$ and $\log_{10} 5 = q$.

\begin{enumerate}[itemsep=0.1cm]
  \item Find an expression for $\log_{10} 45$ in terms of $p$ and $q$.
  \hfill [3]
  \item Find an expression for $\log_{3} 25$ in terms of $p$ and $q$.
  \hfill [2]
\end{enumerate}

  \draw (-0.6,0) rectangle (11,7);
  \foreach \y in {1,2,...,6} \draw[dotted] (0,\y)--(10.5,\y);
\end{tikzpicture}

%--- Problem 3: log_10 b = 2/3 [5 marks] ---
\item[3.] [Maximum mark: 5]

It is given that $\log_{10} b = \dfrac{2}{3}$, where $b > 0$.

Find the value of

\begin{enumerate}[itemsep=0.1cm]
  \item $\log_{10}\!\left(\dfrac{1}{b}\right)$; \hfill [2]
  \item $\log_{1000} b$. \hfill [3]
\end{enumerate}

\medskip

%--- Problem 4: Solve 2 ln x = ln 16 + 4 [5 marks] ---
\item[4.] [Maximum mark: 5]

Solve the equation $2\ln x = \ln 16 + 4$. Give your answer in the
form $x = p\, e^{q}$ where $p, q \in \mathbb{Z}^{+}$.
