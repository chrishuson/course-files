\subsubsection*{7.11 PreQuiz: Logarithmic and Exponential Functions ---
Markscheme}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Quarterly compound interest [6 marks] ---
\item[1.] [Maximum mark: 6]

Lina invests \$2400 at $6\,\%$ per annum, compounded quarterly.

\medskip

\textbf{(a)} Quarterly rate $=\dfrac{0.06}{4}=0.015$, so the quarterly
growth factor is $1.015$. \hfill \textbf{A1}

\emph{Accept $1.5\,\%$ with a feedback note that, for a growth
``factor'', $1.015$ is the preferred form.}

\medskip

\textbf{(b)} There are $5 \times 4 = 20$ quarters. \hfill \textbf{M1}

$V=2400(1.015)^{20}=\$3232.45$ \hfill \textbf{A1}

\medskip

\textbf{(c)} Need the smallest integer $n$ with
$2400(1.015)^{4n}>3400$. \hfill \textbf{M1}

$4n>\dfrac{\ln(3400/2400)}{\ln(1.015)}$, so
$n>5.85\ldots$. \hfill \textbf{A1}

Smallest full-year $n=6$ years. \hfill \textbf{A1}

\emph{Alternative method (table): show both adjacent rows
$2400(1.015)^{20}=3232.45$ (at $n=5$) and $2400(1.015)^{24}=3430.84$
(at $n=6$), then state $n=6$. A single-row table earns full marks but
should receive a feedback note recommending the adjacent row to
justify the crossover.}

\newpage

%--- Problem 2: Continuous exponential growth [6 marks] ---
\item[2.] [Maximum mark: 6]

$M(t)=5000e^{kt}$ and $M(2)=7200$.

\medskip

\textbf{(a)} $5000e^{2k}=7200$, so $e^{2k}=1.44$. \hfill \textbf{M1}

$k=\dfrac{\ln 1.44}{2}=0.1823\ldots$ \hfill \textbf{A1}

\medskip

\textbf{(b)} $M(6)=5000e^{6k}$, using their value of $k$.
\hfill \textbf{M1}

$M(6)=14\,929.92\ldots$, so $14\,930$ members. \hfill \textbf{A1}

\emph{Accept $14\,901$ (obtained by substituting rounded $k=0.182$)
with a feedback note: do not round intermediate values; carry full
precision through and round only at the end.}

\medskip

\textbf{(c)} Set $5000e^{kt}=10\,000$, so $e^{kt}=2$.
\hfill \textbf{M1}

$t=\dfrac{\ln 2}{k}=3.80$ years. \hfill \textbf{A1}

\emph{Accept $3.81$ (obtained by substituting rounded $k=0.182$) with
the same feedback note about not rounding intermediate values.}

\end{enumerate}
