\subsubsection*{7.10 PreQuiz: Logarithmic and Exponential Functions ---
Markscheme}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Reading a transformed log graph [6 marks] ---
\item[1.] [Maximum mark: 6]

The graph of $f(x) = \log_{3}(x+1) + 2$ is shown.

\medskip

\textbf{(a)} Vertical asymptote where the argument of the log is zero:
$x + 1 = 0$. \hfill \textbf{A1}

$x = -1$

\medskip

\textbf{(b)} Set $f(x) = 0$:
$\log_{3}(x + 1) + 2 = 0$, so $\log_{3}(x + 1) = -2$. \hfill \textbf{M1}

$x + 1 = 3^{-2} = \dfrac{1}{9}$ \hfill \textbf{A1}

$x = -\dfrac{8}{9}$ \hfill \textbf{A1}

\emph{Accept $x \approx -0.889$ (3 s.f.).}

\medskip

\textbf{(c)} Range: $f(x) \in \mathbb{R}$ (all real numbers).
\hfill \textbf{A1}

\medskip

\textbf{(d)} $f(8) = \log_{3}(9) + 2 = 2 + 2 = 4$, so the point is
$(8,\, 4)$. \hfill \textbf{A1}

\newpage

%--- Problem 2: Exponential function and its inverse [6 marks] ---
\item[2.] [Maximum mark: 6]

Let $g(x) = 4 \cdot 2^{\,x-1} - 3$.

\medskip

\textbf{(a)} As $x \to -\infty$, $2^{x-1} \to 0$, so $g(x) \to -3$.

Horizontal asymptote: $y = -3$. \hfill \textbf{A1}

\medskip

\textbf{(b)} $g(3) = 4 \cdot 2^{\,3-1} - 3 = 4 \cdot 4 - 3$
\hfill \textbf{M1}

$= 13$ \hfill \textbf{A1}

\medskip

\textbf{(c)} Let $y = 4 \cdot 2^{\,x-1} - 3$. Interchange $x$ and $y$
and solve for $y$:

$x = 4 \cdot 2^{\,y - 1} - 3$

$x + 3 = 4 \cdot 2^{\,y - 1}$ \hfill \textbf{M1}

$2^{\,y - 1} = \dfrac{x + 3}{4}$

Take $\log_{2}$ of both sides:
$y - 1 = \log_{2}\!\left(\dfrac{x+3}{4}\right)$ \hfill \textbf{A1}

$g^{-1}(x) = \log_{2}\!\left(\dfrac{x + 3}{4}\right) + 1$
\hfill \textbf{A1}

\emph{Accept equivalent forms, e.g.\
$g^{-1}(x) = \log_{2}(x + 3) - 1$.}

\newpage

%--- Problem 3: Continuous exponential growth [6 marks] ---
\item[3.] [Maximum mark: 6]

$N(t) = 8000\, e^{\,kt}$, with $N(3) = 24\,000$.

\medskip

\textbf{(a)} $8000\, e^{\,3k} = 24\,000$, so $e^{\,3k} = 3$.
\hfill \textbf{M1}

$3k = \ln 3$, hence $k = \dfrac{1}{3}\ln 3$ \quad ($a = 3$).
\hfill \textbf{A1}

\medskip

\textbf{(b)} $N(7) = 8000\, e^{\,7k} = 8000 \cdot 3^{7/3}$.

\emph{GDC: } $8000 \times e^{(7/3)\ln 3} = 8000 \times 12.9805\ldots$
\hfill \textbf{M1}

$N(7) \approx 103\,844$ users. \hfill \textbf{A1}

\medskip

\textbf{(c)} Set $8000\, e^{\,kt} = 72\,000$, so $e^{\,kt} = 9$.

$kt = \ln 9 = 2\ln 3$ \hfill \textbf{M1}

$t = \dfrac{2 \ln 3}{\frac{1}{3}\ln 3} = 6$ years. \hfill \textbf{A1}

\emph{Accept $t \approx 6.00$ (3 s.f.) or by GDC solving
$8000\, e^{(t \ln 3)/3} = 72000$.}

\newpage

%--- Problem 4: Compound interest [6 marks] ---
\item[4.] [Maximum mark: 6]

Sofia invests \$2500 at $4\,\%$ per annum, compounded annually.

\medskip

\textbf{(a)} End-of-year-1 value $= 2500 \times 1.04 = 2600$. \hfill \textbf{AG}

\medskip

\textbf{(b)} $V_{10} = 2500 \times 1.04^{\,10}$ \hfill \textbf{M1}

$= 2500 \times 1.48024\ldots$

$= \$3700.61$ (nearest cent). \hfill \textbf{A1}

\medskip

\textbf{(c)} Need the smallest integer $n$ with
$2500 \times 1.04^{\,n} \ge 5000$, i.e.\ $1.04^{\,n} \ge 2$.
\hfill \textbf{M1}

Take logs: $n \ge \dfrac{\ln 2}{\ln 1.04} = 17.673\ldots$
\hfill \textbf{A1}

Smallest full-year $n = 18$ years. \hfill \textbf{A1}

\emph{GDC check: $1.04^{17} = 1.948\ldots$ (under), $1.04^{18} =
2.026\ldots$ (over); equivalently $2500 \times 1.04^{17} = 4870.38$
and $2500 \times 1.04^{18} = 5065.20$.}

\newpage

%--- Problem 5: Logarithmic scale (Richter) [5 marks] ---
\item[5.] [Maximum mark: 5]

$M = \log_{10}\!\left(\dfrac{I}{I_{0}}\right)$.

\medskip

\textbf{(a)} $5.4 = \log_{10}\!\left(\dfrac{I}{I_{0}}\right)$, so
$\dfrac{I}{I_{0}} = 10^{\,5.4}$. \hfill \textbf{M1}

$= 251\,188.6\ldots = 2.51 \times 10^{5}$ (3 s.f.).
\hfill \textbf{A1}

\medskip

\textbf{(b)} $I_{B} = 500\, I_{A}$, so

$M_{B} = \log_{10}\!\left(\dfrac{500\, I_{A}}{I_{0}}\right)
= \log_{10}(500) + \log_{10}\!\left(\dfrac{I_{A}}{I_{0}}\right)$
\hfill \textbf{M1}

$= \log_{10}(500) + 5.4$ \hfill \textbf{A1}

$= 2.69897\ldots + 5.4 = 8.09897\ldots$

$M_{B} \approx 8.10$ (3 s.f.). \hfill \textbf{A1}

\emph{Accept any equivalent method, e.g.\
$M_{B} = \log_{10}(500 \times 10^{5.4})$ evaluated by GDC.}

\newpage

%--- Problem 6: Light absorption (exponential decay) [4 marks] ---
\item[6.] [Maximum mark: 4]

$L(d) = L_{0}\, e^{-0.32\, d}$.

\medskip

\textbf{(a)} $\dfrac{L(4)}{L_{0}} = e^{-0.32 \times 4} = e^{-1.28}$.
\hfill \textbf{M1}

$= 0.27804\ldots \approx 27.8\,\%$ (3 s.f.). \hfill \textbf{A1}

\medskip

\textbf{(b)} Set $e^{-0.32\, d} = \dfrac{1}{2}$.

$-0.32\, d = \ln \dfrac{1}{2} = -\ln 2$ \hfill \textbf{M1}

$d = \dfrac{\ln 2}{0.32} = 2.1660\ldots \approx 2.17$ m (3 s.f.).
\hfill \textbf{A1}

\end{enumerate}
