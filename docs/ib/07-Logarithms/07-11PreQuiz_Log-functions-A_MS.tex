\subsubsection*{7.11 PreQuiz: Logarithmic and Exponential Functions ---
Markscheme}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Compound interest [6 marks] ---
\item[1.] [Maximum mark: 6]

Maya invests \$1800 at $5\,\%$ per annum, compounded annually.

\medskip

\textbf{(a)} End-of-year-1 value $=1800 \times 1.05 = 1890$.
\hfill \textbf{AG}

\medskip

\textbf{(b)} $V_8 = 1800(1.05)^8$ \hfill \textbf{M1}

$V_8 = \$2659.42$ \hfill \textbf{A1}

\medskip

\textbf{(c)} Need the smallest integer $n$ with
$1800(1.05)^n > 2700$. \hfill \textbf{M1}

$n > \dfrac{\ln(2700/1800)}{\ln(1.05)} = 8.31\ldots$
\hfill \textbf{A1}

Smallest full-year $n=9$ years. \hfill \textbf{A1}

\newpage

%--- Problem 2: Continuous exponential growth [6 marks] ---
\item[2.] [Maximum mark: 6]

$F(t)=1200e^{0.18t}$.

\medskip

\textbf{(a)} $F(1)=1200e^{0.18}=1436.66\ldots$, so about
$1437$ followers. \hfill \textbf{A1}

\medskip

\textbf{(b)} $F(4)=1200e^{0.72}$ \hfill \textbf{M1}

$F(4)=2465.32\ldots$, so $2465$ followers. \hfill \textbf{A1}

\medskip

\textbf{(c)} Set $1200e^{0.18t}=3000$. \hfill \textbf{M1}

$e^{0.18t}=2.5$, so $0.18t=\ln 2.5$. \hfill \textbf{M1}

$t=\dfrac{\ln 2.5}{0.18}=5.09$ years. \hfill \textbf{A1}

\end{enumerate}
