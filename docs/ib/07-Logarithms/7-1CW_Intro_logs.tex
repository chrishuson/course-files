\subsubsection*{7.1 Classwork: Introduction to Logarithms (no calculator)}

\begin{enumerate}[itemsep=0.25cm]

%--- Problem 1: Evaluate basic logarithms [12 marks] ---
%--- Source: adapted from 5.15 Q1(a) and 5.16 Q1(a) ---
\item[1.] Evaluate each of the following.

\begin{enumerate}[label=(\alph*), itemsep=1cm]
  \item $\log_4 \dfrac{1}{64}$
  \vspace{0.9cm}

  \item $\log_{25} 5$
  \vspace{0.9cm}

  \item $\log_2 \sqrt{32}$
  \vspace{0.9cm}

  \item $\log_5 125$
  \vspace{0.9cm}

  \item $\log_{16} \dfrac{1}{4}$
  \vspace{0.9cm}

  \item $\log_3 \sqrt{81}$
  \vspace{0.9cm}
\end{enumerate}

\newpage

%--- Problem 2: Write logarithmic expressions in terms of c [12 marks] ---
%--- Source: adapted from 5.15 Q1(b) and 5.16 Q1(b) ---
\item[2.] Let $c = \log_3 m$, where $m > 0$. In each question, write the expression in terms of $c$.

\begin{enumerate}[label=(\alph*), itemsep=1cm]
  \item $\log_3 m^2$
  \vspace{1cm}

  \item $\log_3(27m)$
  \vspace{1cm}

  \item $\log_9 m$
  \vspace{2cm} \\
Let $a = \log_2 m$, where $m > 0$. Write the expression in terms of $a$.

  \item $\log_2 m^3$
  \vspace{1cm}

  \item $\log_2(8m)$
  \vspace{1cm}

  \item $\log_4 m$
  \vspace{1cm}
\end{enumerate}

\newpage

%--- Problem 3: Expand and condense logarithmic expressions [12 marks] ---
%--- Source: adapted from 5.16 Q2 ---
\item[3.] Use the laws of logarithms.

\begin{enumerate}[itemsep=1.5cm]
  \item Express each in terms of $\log x$ and $\log y$.
  \begin{enumerate}[itemsep=1cm, label=(\roman*)]
    \item $\log (100x)$
    \item $\log \!\left(\dfrac{x^2}{y}\right)$
    \item $\log \sqrt{xy^3}$ \hfill [6]
  \end{enumerate}

  \item Write each expression as a single logarithm.
  \begin{enumerate}[label=(\roman*), itemsep=0.8cm]
    \item $\log_2 x + \log_2 y$
    \vspace{0.6cm}

    \item $2\log_7 a - \log_7 b$
    \vspace{0.6cm}

    \item $\dfrac{1}{2}\ln p + \ln q$
    \vspace{0.6cm}
  \end{enumerate}
\end{enumerate}

\newpage

%--- Problem 4: Solve basic logarithmic equations [12 marks] ---
%--- Source: adapted from 5.15 Q2 and introductory classwork style ---
\item[4.] Solve each equation. Give exact answers.

\begin{enumerate}[label=(\alph*), itemsep=1cm]
  \item $\log_2(x - 3) = \log_4(x + 9)$
  \vspace{1.6cm}

  \item $\log_3 x = 2$
  \vspace{1.6cm}

  \item $\log_5(2x) = 1$
  \vspace{1.6cm}

  \item $\log_2(x+1) + \log_2(x-1) = 3$
  \vspace{1.6cm}
\end{enumerate}

\end{enumerate}
