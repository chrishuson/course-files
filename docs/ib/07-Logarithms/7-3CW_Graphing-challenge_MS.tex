\subsubsection*{7.3 Classwork: Logarithm and Graphing Challenge --- Markscheme}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Change of base [9 marks] ---
\item[1.] [Maximum mark: 9]

\textbf{(a)} Solve $\log_2 x + \log_4 x + \log_8 x = 11$.

Change all terms to base $2$:
$\log_4 x = \dfrac{\log_2 x}{\log_2 4} = \dfrac{1}{2}\log_2 x$ \;and\;
$\log_8 x = \dfrac{1}{3}\log_2 x$ \hfill \textbf{M1}

$\log_2 x + \dfrac{1}{2}\log_2 x + \dfrac{1}{3}\log_2 x = 11$ \hfill \textbf{A1}

$\dfrac{11}{6}\log_2 x = 11 \;\Rightarrow\; \log_2 x = 6$ \hfill \textbf{A1}

$x = 2^{6} = 64$ \hfill \textbf{A1}

\medskip

\textbf{(b)} Show that $\log_a b \cdot \log_b a = 1$.

Using change of base, $\log_a b = \dfrac{\ln b}{\ln a}$ and
$\log_b a = \dfrac{\ln a}{\ln b}$. \hfill \textbf{M1}

$\log_a b \cdot \log_b a
= \dfrac{\ln b}{\ln a} \cdot \dfrac{\ln a}{\ln b} = 1$ \hfill \textbf{A1\;AG}

\medskip

\textbf{(c)} Given $\log_a b = 4$, find $\log_{a^2} b^3$.

$\log_{a^2} b^3 = \dfrac{\log_a b^3}{\log_a a^2}
= \dfrac{3 \log_a b}{2}$ \hfill \textbf{M1\;A1}

$= \dfrac{3}{2}\cdot 4 = 6$ \hfill \textbf{A1}

\newpage

%--- Problem 2: Quadratic in log [5 marks] ---
\item[2.] [Maximum mark: 5]

Solve $(\log_3 x)^2 - 5 \log_3 x + 6 = 0$.

\textbf{Solution:}

Let $u = \log_3 x$, so $u^2 - 5u + 6 = 0$. \hfill \textbf{M1}

$(u - 2)(u - 3) = 0$ \hfill \textbf{A1}

$u = 2$ \;\;or\;\; $u = 3$ \hfill \textbf{A1}

\medskip

Back-substitute:
\;\;$\log_3 x = 2 \;\Rightarrow\; x = 3^2 = 9$ \hfill \textbf{A1}

\;\;$\log_3 x = 3 \;\Rightarrow\; x = 3^3 = 27$ \hfill \textbf{A1}

\smallskip

(Both solutions are valid since each gives $x > 0$.)

\newpage

%--- Problem 3: Exponential and its log inverse [9 marks] ---
\item[3.] [Maximum mark: 9]

Let $f(x) = 3^{\,x-1} + 2$.

\textbf{(a)} Find $f^{-1}(x)$.

Set $y = 3^{\,x-1} + 2$ and solve for $x$:

$y - 2 = 3^{\,x-1}$ \hfill \textbf{M1}

$\log_3(y - 2) = x - 1$ \hfill \textbf{A1}

$x = 1 + \log_3(y - 2)$ \hfill \textbf{A1}

$f^{-1}(x) = 1 + \log_3(x - 2)$ \hfill \textbf{A1}

\medskip

\textbf{(b)} Sketch of $y = f(x)$, $y = f^{-1}(x)$, and $y = x$:

\begin{itemize}\setlength\itemsep{0pt}
  \item Correct sketch of $f$ (increasing exponential through $(1, 3)$,
  rising to upper-right) \hfill \textbf{A1}
  \item Horizontal asymptote $y = 2$ drawn and labelled \hfill \textbf{A1}
  \item Correct sketch of $f^{-1}$ (reflection of $f$ across $y = x$;
  passes through $(3, 1)$) \hfill \textbf{A1}
  \item Vertical asymptote $x = 2$ drawn and labelled \hfill \textbf{A1}
  \item Line $y = x$ drawn \hfill \textbf{A1}
\end{itemize}

\noindent\emph{Reference points used to verify the sketches:}
\;$f(0) = \tfrac{1}{3} + 2 \approx 2.33$, \;$f(1) = 3$, \;$f(2) = 5$;\;
correspondingly $f^{-1}(2.33) \approx 0$, \;$f^{-1}(3) = 1$,
\;$f^{-1}(5) = 2$.

\newpage

%--- Problem 4: Solve for the exponent [12 marks] ---
\item[4.] [Maximum mark: 12]

\textbf{(a)} Solve $2^x = 20$.

Take $\log_2$ of both sides: \hfill \textbf{M1}

$x = \log_2 20$ \hfill \textbf{A1}

\smallskip

\emph{Equivalent forms accepted:} $\dfrac{\ln 20}{\ln 2}$,
$\dfrac{\log 20}{\log 2}$, $2 + \log_2 5$.

\medskip

\textbf{(b)} Solve $5^{\,x-1} = 12$.

Take $\log_5$ of both sides: \hfill \textbf{M1}

$x - 1 = \log_5 12$ \hfill \textbf{A1}

$x = 1 + \log_5 12$ \hfill \textbf{A1}

\smallskip

\emph{Equivalent form:} $x = 1 + \dfrac{\ln 12}{\ln 5}$.

\medskip

\textbf{(c)} $P(t) = 50 \cdot 2^{\,t}$; find $t$ such that $P = 1500$.

$50 \cdot 2^{\,t} = 1500 \;\Rightarrow\; 2^{\,t} = 30$ \hfill \textbf{M1\;A1}

$t = \log_2 30$ \hfill \textbf{A1}

\smallskip

\emph{Equivalent forms:} $\dfrac{\ln 30}{\ln 2}$,
$1 + \log_2 15$.

\medskip

\textbf{(d)} Solve $5^{\,2x} = 3^{\,x+1}$.

Take $\ln$ of both sides: \hfill \textbf{M1}

$2x \ln 5 = (x + 1) \ln 3$ \hfill \textbf{A1}

Collect $x$-terms:
$2x \ln 5 - x \ln 3 = \ln 3$, \;so\; $x(2\ln 5 - \ln 3) = \ln 3$
\hfill \textbf{M1}

$x = \dfrac{\ln 3}{2 \ln 5 - \ln 3} = \dfrac{\ln 3}{\ln(25/3)}$
\hfill \textbf{A1}

\end{enumerate}
