%--- Problem 2: Log series geometric/arithmetic [15 marks] ---
%--- Source: AA SL 2022 May P1 TZ1 Q08 ---
\item[2.] [Maximum mark: 15]

\begingroup
\setlength{\parskip}{0pt}
\setlength{\parsep}{0pt}

Consider the series $\ln x + p \ln x + \dfrac{1}{3} \ln x + \ldots$,
where $x \in \mathbb{R}$, $x > 1$ and $p \in \mathbb{R}$, $p \neq 0$.

\begin{enumerate}[itemsep=0.05cm, topsep=0.05cm]
  \item Consider the case where the series is geometric.
  \begin{enumerate}[itemsep=0.02cm, topsep=0.02cm, label=(\roman*)]
    \item Show that $p = \pm \dfrac{1}{\sqrt{3}}$.
    \item Given that $p > 0$ and $S_{\infty} = 3 + \sqrt{3}$,
    find the value of $x$. \hfill [5]
  \end{enumerate}

  \item Now consider the case where the series is arithmetic with common
  difference $d$.
  \begin{enumerate}[itemsep=0.02cm, topsep=0.02cm, label=(\roman*)]
    \item Show that $p = \dfrac{2}{3}$.
    \item Write down $d$ in the form $k \ln x$, where $k \in \mathbb{R}$.
    \item The sum of the first $n$ terms of the series is $-3 \ln x$.
    Find the value of $n$. \hfill [10]
  \end{enumerate}
\end{enumerate}
\endgroup
