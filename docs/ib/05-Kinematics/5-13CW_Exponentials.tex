\begin{enumerate}[itemsep=0.15cm]
  \item Show that $\dfrac{\mathrm{d}y}{\mathrm{d}x} = \dfrac{1 - 4\ln x}{x^5}$. \hfill [3]
\end{enumerate}

Consider the function defined by $f(x) = \dfrac{\ln x}{x^4}$ for $x > 0$
and its graph $y = f(x)$.

\begin{enumerate}[itemsep=0.15cm]
  \item[(b)] The graph of $f$ has a horizontal tangent at point $P$.
  Find the coordinates of $P$. \hfill [5]
  \item[(c)] Given that $f''(x) = \dfrac{20\ln x - 9}{x^6}$, show that $P$
  is a local maximum point. \hfill [3]
  \item[(d)] Solve $f(x) > 0$ for $x > 0$. \hfill [2]
  \item[(e)] Sketch the graph of $f$, showing clearly the value of the
  $x$-intercept and the approximate position of point $P$. \hfill [3]
\end{enumerate}
