%--- Problem 2: Write logarithmic expressions in terms of a parameter [12 marks] ---
%--- Source: adapted from 7-1CW_Intro_logs Q2 ---
\item[2.] [Maximum mark: 12]

\begin{enumerate}[itemsep=0.15cm]
  \item Let $c = \log_3 m$, where $m > 0$. Write each of the
  following expressions in terms of $c$.
  \begin{enumerate}[itemsep=0.1cm, label=(\roman*)]
    \item $\log_3 m^2$
    \item $\log_3(27m)$
    \item $\log_9 m$ \hfill [6]
  \end{enumerate}

  \item Let $a = \log_2 m$, where $m > 0$. Write each of the
  following expressions in terms of $a$.
  \begin{enumerate}[itemsep=0.1cm, label=(\roman*)]
    \item $\log_2 m^3$
    \item $\log_2(8m)$
    \item $\log_4 m$ \hfill [6]
  \end{enumerate}
\end{enumerate}

%--- Problem 3: Expand and condense logarithmic expressions [12 marks] ---
%--- Source: adapted from 7-1CW_Intro_logs Q3 ---
\item[3.] [Maximum mark: 12]

Use the laws of logarithms.

\begin{enumerate}[itemsep=0.15cm]
  \item Express each in terms of $\log x$ and $\log y$.
  \begin{enumerate}[itemsep=0.1cm, label=(\roman*)]
    \item $\log (100x)$
    \item $\log \!\left(\dfrac{x^2}{y}\right)$
    \item $\log \sqrt{xy^3}$ \hfill [6]
  \end{enumerate}

  \item Write each expression as a single logarithm.
  \begin{enumerate}[itemsep=0.1cm, label=(\roman*)]
    \item $\log_2 x + \log_2 y$
    \item $2\log_7 a - \log_7 b$
    \item $\dfrac{1}{2}\ln p + \ln q$ \hfill [6]
  \end{enumerate}
\end{enumerate}

%--- Problem 4: Solve basic logarithmic equations [12 marks] ---

\item[4.] [Maximum mark: 12]

Solve each equation. Give exact answers.

\begin{enumerate}[itemsep=0.1cm]
  \item $\log_2(x - 3) = \log_4(x + 9)$
  \item $\log_3 x = 2$
  \item $\log_5(2x) = 1$
  \item $\log_2(x+1) + \log_2(x-1) = 3$ \hfill [12]
\end{enumerate}
