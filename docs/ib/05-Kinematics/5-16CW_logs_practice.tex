\subsubsection*{5.16 Classwork: Logarithms Practice (no calculator)}

\begin{enumerate}[itemsep=0.15cm]

%--- Problem 1: Evaluate logarithms and write expressions in terms of c [12 marks] ---
%--- Source: original, modeled on recent classwork problems on basic logarithm rules ---
\item[1.] [Maximum mark: 12]

\begin{enumerate}[itemsep=0.2cm]
  \item Calculate the value of each of the following logarithms:
  \begin{enumerate}[itemsep=0.1cm, label=(\roman*)]
    \item $\log_5 125$
    \item $\log_{16} \dfrac{1}{4}$
    \item $\log_3 \sqrt{81}$ \hfill [6]
  \end{enumerate}

  \item Let $u = \log_2 x$, where $x > 0$. Write down each of the
  following expressions in terms of $u$.
  \begin{enumerate}[itemsep=0.1cm, label=(\roman*)]
    \item $\log_2 x^3$
    \item $\log_2(8x)$
    \item $\log_4 x$ \hfill [6]
  \end{enumerate}
\end{enumerate}

%--- Problem 2: Expand and condense logarithmic expressions [12 marks] ---
%--- Source: original, modeled on standard IB-style practice with logarithm laws ---
\item[2.] [Maximum mark: 12]

\begin{enumerate}[itemsep=0.2cm]
  \item Express each in terms of $\log x$ and $\log y$.
  \begin{enumerate}[itemsep=0.1cm, label=(\roman*)]
    \item $\log (100x)$
    \item $\log \!\left(\dfrac{x^2}{y}\right)$
    \item $\log \sqrt{xy^3}$ \hfill [6]
  \end{enumerate}

  \item Write each of the following as a single logarithm.
  \begin{enumerate}[itemsep=0.1cm, label=(\roman*)]
    \item $\log_2 x + \log_2 y$
    \item $2\log_7 a - \log_7 b$
    \item $\dfrac{1}{2}\ln p + \ln q$ \hfill [6]
  \end{enumerate}
\end{enumerate}
