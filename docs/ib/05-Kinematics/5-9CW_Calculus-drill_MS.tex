\subsubsection*{5.9 Classwork: Differentiation and Integration Drill --- Markscheme}

\begin{enumerate}[itemsep=0.4cm]

\textbf{--- Differentiation ---}

\item $f(x) = 4x^3 - 6x + 7$

$f'(x) = 12x^2 - 6$ \hfill \textbf{(A1)}

\item $y = 5\sin x - 2\cos x$

$\dfrac{\mathrm{d}y}{\mathrm{d}x} = 5\cos x + 2\sin x$ \hfill \textbf{(A1)(A1)}

\item $f(x) = 3e^{2x}$

$f'(x) = 3 \cdot 2e^{2x} = 6e^{2x}$ \hfill \textbf{(M1)(A1)}

\item $y = 4\ln x + 3x^{-1}$

$\dfrac{\mathrm{d}y}{\mathrm{d}x} = \dfrac{4}{x} - \dfrac{3}{x^2}$
\hfill \textbf{(A1)(A1)}

\item $f(x) = \sin(3x + 1)$ \hfill \emph{chain rule}

$f'(x) = 3\cos(3x + 1)$ \hfill \textbf{(M1)(A1)}

\item $y = e^{x^2 - 1}$ \hfill \emph{chain rule}

$\dfrac{\mathrm{d}y}{\mathrm{d}x} = 2x \cdot e^{x^2 - 1}$
\hfill \textbf{(M1)(A1)}

\item $f(x) = x^2 \ln x$ \hfill \emph{product rule}

$f'(x) = 2x \cdot \ln x + x^2 \cdot \dfrac{1}{x} = 2x\ln x + x = x(2\ln x + 1)$
\hfill \textbf{(M1)(A1)(A1)}

\bigskip
\textbf{--- Integration ---}

\item $\displaystyle\int (6x^2 + 4x - 1)\,\mathrm{d}x = 2x^3 + 2x^2 - x + c$
\hfill \textbf{(A1)(A1)}

\item $\displaystyle\int \!\left(3\cos x + \frac{1}{x}\right)\mathrm{d}x
= 3\sin x + \ln x + c$ \quad ($x > 0$)
\hfill \textbf{(A1)(A1)}

\item $\displaystyle\int 5e^{3x}\,\mathrm{d}x = \frac{5}{3}e^{3x} + c$
\hfill \textbf{(M1)(A1)}

\item $\displaystyle\int_0^{\pi} 2\sin x\,\mathrm{d}x
= \bigl[-2\cos x\bigr]_0^{\pi}$
\hfill \textbf{(M1)}

$= -2\cos\pi - (-2\cos 0) = -2(-1) + 2(1) = 4$
\hfill \textbf{(A1)(A1)}

\item $\displaystyle\int_1^2 \!\left(\frac{3}{x} + 2x\right)\mathrm{d}x
= \bigl[3\ln x + x^2\bigr]_1^2$
\hfill \textbf{(M1)}

$= (3\ln 2 + 4) - (0 + 1) = 3\ln 2 + 3$
\hfill \textbf{(A1)(A1)}

\item $f'(x) = 4x + e^x$, \quad $f(0) = 3$

$f(x) = 2x^2 + e^x + c$ \hfill \textbf{(A1)}

$f(0) = 0 + 1 + c = 3 \Rightarrow c = 2$ \hfill \textbf{(M1)}

$f(x) = 2x^2 + e^x + 2$ \hfill \textbf{(A1)}

\item $f(x) = (2x + 1)e^x$

$f'(x) = 2e^x + (2x + 1)e^x = (2x + 3)e^x$ \hfill \textbf{(M1)(A1)}

Hence $\displaystyle\int (2x + 3)e^x\,\mathrm{d}x = (2x + 1)e^x + c$
\hfill \textbf{(M1)(A1)}

\end{enumerate}
