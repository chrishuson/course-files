\subsubsection*{11.14 Classwork: Calculus practice (no calculator) --- Markscheme}

\textbf{Part A. Differentiate.}

\begin{enumerate}[itemsep=0.3cm]

%--- 1. y = (3x^2 + 1)^4 ---
\item[1.] $y = (3x^2 + 1)^4$

\medskip

Outer derivative: $4(3x^2 + 1)^3$. \quad Inner derivative: $\dfrac{d}{dx}(3x^2 + 1) = 6x$. \hfill \textbf{M1}

$\dfrac{dy}{dx} = 4(3x^2 + 1)^3 \cdot 6x = 24x(3x^2 + 1)^3$ \hfill \textbf{A1}

%--- 2. y = sqrt(2x^2 + 1) ---
\item[2.] $y = \sqrt{2x^2 + 1} = (2x^2 + 1)^{1/2}$

\medskip

$\dfrac{dy}{dx} = \dfrac{1}{2}(2x^2 + 1)^{-1/2} \cdot 4x$ \hfill \textbf{M1}

$= \dfrac{2x}{\sqrt{2x^2 + 1}}$ \hfill \textbf{A1}

%--- 3. y = sin(2x^3) ---
\item[3.] $y = \sin(2x^3)$

\medskip

$\dfrac{dy}{dx} = \cos(2x^3) \cdot 6x^2$ \hfill \textbf{M1}

$= 6x^2 \cos(2x^3)$ \hfill \textbf{A1}

%--- 4. y = e^(x^2 - 4x) ---
\item[4.] $y = e^{x^2 - 4x}$

\medskip

$\dfrac{dy}{dx} = e^{x^2 - 4x} \cdot (2x - 4)$ \hfill \textbf{M1}

$= (2x - 4) e^{x^2 - 4x}$ \hfill \textbf{A1}

%--- 5. y = ln(5x^2 + 3) ---
\item[5.] $y = \ln(5x^2 + 3)$

\medskip

$\dfrac{dy}{dx} = \dfrac{1}{5x^2 + 3} \cdot 10x$ \hfill \textbf{M1}

$= \dfrac{10x}{5x^2 + 3}$ \hfill \textbf{A1}

%--- 6. y = cos^3(2x) — chain rule twice ---
\item[6.] $y = \cos^3(2x) = [\cos(2x)]^3$

\medskip

Outer: $3[\cos(2x)]^2$. \quad Then differentiate $\cos(2x)$: $-\sin(2x) \cdot 2 = -2\sin(2x)$. \hfill \textbf{M1 M1}

$\dfrac{dy}{dx} = 3\cos^2(2x) \cdot (-2\sin(2x)) = -6\cos^2(2x)\sin(2x)$ \hfill \textbf{A1}

\end{enumerate}

\textbf{Part B. Integrate.} \quad For each, identify a substitution $u$, integrate, then apply the initial condition to find $C$.

\begin{enumerate}[itemsep=0.3cm, resume]

%--- 7. f'(x) = 6x / sqrt(x^2 + 4), f(0)=5 ---
\item[7.] $f'(x) = \dfrac{6x}{\sqrt{x^2 + 4}}$, $\quad f(0) = 5$

\medskip

Let $u = x^2 + 4$, so $du = 2x\,dx$ and $6x\,dx = 3\,du$. \hfill \textbf{M1}

$\displaystyle f(x) = \int \dfrac{3}{\sqrt{u}}\,du = 6\sqrt{u} + C = 6\sqrt{x^2 + 4} + C$ \hfill \textbf{A1}

Apply $f(0) = 5$: $\quad 6\sqrt{4} + C = 5 \;\Rightarrow\; 12 + C = 5 \;\Rightarrow\; C = -7$ \hfill \textbf{M1 A1}

$f(x) = 6\sqrt{x^2 + 4} - 7$ \hfill \textbf{A1}

%--- 8. f'(x) = 2x cos(x^2), f(0)=1 ---
\item[8.] $f'(x) = 2x \cos(x^2)$, $\quad f(0) = 1$

\medskip

Let $u = x^2$, so $du = 2x\,dx$. \hfill \textbf{M1}

$\displaystyle f(x) = \int \cos u\,du = \sin u + C = \sin(x^2) + C$ \hfill \textbf{A1}

Apply $f(0) = 1$: $\quad \sin 0 + C = 1 \;\Rightarrow\; C = 1$ \hfill \textbf{M1}

$f(x) = \sin(x^2) + 1$ \hfill \textbf{A1}

%--- 9. f'(x) = 2x / (x^2+1), f(1)=0 ---
\item[9.] $f'(x) = \dfrac{2x}{x^2 + 1}$, $\quad f(1) = 0$

\medskip

Let $u = x^2 + 1$, so $du = 2x\,dx$. \hfill \textbf{M1}

$\displaystyle f(x) = \int \dfrac{1}{u}\,du = \ln|u| + C = \ln(x^2 + 1) + C$ \hfill \textbf{A1}

Apply $f(1) = 0$: $\quad \ln 2 + C = 0 \;\Rightarrow\; C = -\ln 2$ \hfill \textbf{M1}

$f(x) = \ln(x^2 + 1) - \ln 2 = \ln\!\left(\dfrac{x^2 + 1}{2}\right)$ \hfill \textbf{A1}

%--- 10. f'(x) = ln(x)/x, f(1)=3 ---
\item[10.] $f'(x) = \dfrac{\ln x}{x}$, $\quad f(1) = 3$

\medskip

Let $u = \ln x$, so $du = \dfrac{1}{x}\,dx$. \hfill \textbf{M1}

$\displaystyle f(x) = \int u\,du = \dfrac{u^2}{2} + C = \dfrac{(\ln x)^2}{2} + C$ \hfill \textbf{A1}

Apply $f(1) = 3$: $\quad 0 + C = 3 \;\Rightarrow\; C = 3$ \hfill \textbf{M1}

$f(x) = \dfrac{(\ln x)^2}{2} + 3$ \hfill \textbf{A1}

\end{enumerate}
