%% PROBLEM 4 %%
\item Two buildings with $PA = 150$, $PB = 90$, $AB = 154$.

(a) Cosine rule: \hfill \textbf{(M1)}

$154^2 = 150^2 + 90^2 - 2(150)(90)\cos(\angle APB)$ \hfill \textbf{(A1)}

$\cos(\angle APB) = \dfrac{150^2 + 90^2 - 154^2}{2(150)(90)}
= \dfrac{22500 + 8100 - 23716}{27000} = \dfrac{6884}{27000}$

$\angle APB = 75.2286\ldots^\circ$

$\angle APB = 75.2^\circ$ (or $1.31$ radians) \hfill \textbf{(A1)}

\medskip
(b) Since both angles of elevation are $\theta$: \hfill \textbf{(M1)}

$\theta = \dfrac{180^\circ - 75.2286\ldots^\circ}{2} = 52.3856\ldots^\circ$ \hfill \textbf{(M1)}

$h = 150\sin\theta = 150\sin(52.3856\ldots^\circ) = 118.820\ldots$

$h = 119$ m \hfill \textbf{(A1)}

\hfill Total [6 marks]

%% PROBLEM 6 %%
\item Drug decay: $A(t) = 500e^{-kt}$.

(a) $A(0) = 500$ mg \hfill \textbf{(A1)}

\medskip
(b) $280 = 500e^{-3k}$ \hfill \textbf{(A1)}

$k = -\dfrac{1}{3}\ln\!\left(\dfrac{280}{500}\right) = 0.193272\ldots$

$k = 0.193$ \hfill \textbf{(A1)}

\medskip
(c) $500e^{-0.193272\ldots T} = 140$ \hfill \textbf{(A1)}

$T = -\dfrac{1}{0.193272}\ln\!\left(\dfrac{140}{500}\right) = 6.58636\ldots$

$T = 6.59$ hours \hfill \textbf{(A1)}

\hfill Total [5 marks]

%% PROBLEM 7 %%
\item Carbon-14: $A = A_0 e^{-kt}$, 100 units at death, half-life 5730 years.

(a) At $t = 0$: $A = A_0 e^{0} = A_0 = 100$ \hfill \textbf{(A1)(AG)}

\medskip
(b) $50 = 100e^{-5730k}$ \hfill \textbf{(M1)}

$e^{-5730k} = \dfrac{1}{2}$

$-5730k = \ln\dfrac{1}{2} = -\ln 2$ \hfill \textbf{(A1)}

$k = \dfrac{\ln 2}{5730}$ \hfill \textbf{(A1)(AG)}

\medskip
(c) 25\% has decayed $\Rightarrow$ 75\% remains, i.e.\ $A = 75$ \hfill \textbf{(A1)}

$75 = 100e^{-(\ln 2/5730)\,t}$ \hfill \textbf{(M1)}

$t = -\dfrac{5730}{\ln 2}\ln(0.75) = 2378.16\ldots$

$t = 2380$ years (to the nearest 10) \hfill \textbf{(A1)}

\hfill Total [7 marks]
