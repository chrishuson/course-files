%--- Problem 1: Linear Regression [7 marks] ---
\item[1.] [Maximum mark: 7]

A sports scientist is studying the relationship between the age of an athlete, $a$ years,
and their reaction time, $r$ milliseconds. The following table shows the data collected
from eight athletes.

\renewcommand{\arraystretch}{1.2}
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|c|c|c|}
\hline
$a$ & 18 & 21 & 25 & 28 & 33 & 36 & 40 & 45 \\
\hline
$r$ & 155 & 159 & 176 & 178 & 198 & 199 & 215 & 226 \\
\hline
\end{tabular}
\end{center}
\renewcommand{\arraystretch}{1.0}

\begin{enumerate}[itemsep=0.3cm]
  \item[(a)] Find the regression line of $r$ on $a$ in the form $r = ma + c$,
  where $m$ and $c$ are constants.
  Give $m$ and $c$ correct to three significant figures. \hfill [3]

  \textit{GDC: Enter data into lists, run linear regression.}

  $m = 2.70$ \hfill \textbf{(A1)}

  $c = 105$ \hfill \textbf{(A1)}

  $r = 2.70a + 105$ \hfill \textbf{(A1)}

  \textit{Award \textbf{(A1)} for each of $m$ and $c$ correct to 3 s.f.,
  and \textbf{(A1)} for writing the equation in the required form.
  Accept $m = 2.70$ (3 s.f.)\ and $c = 105$ (3 s.f.).}

  \item[(b)] Write down the value of Pearson's product-moment correlation
  coefficient, $r$. \hfill [1]

  $r = 0.994$ \hfill \textbf{(A1)}

  \textit{Award for $r = 0.994$ (3 s.f.). Full GDC display acceptable as working.}

  \item[(c)] Use the regression line to estimate the reaction time
  of an athlete who is 30 years old. \hfill [2]

  $r = 2.70(30) + 105$ \hfill \textbf{(M1)}

  $= 186$ ms \hfill \textbf{(A1)}

  \textit{Award \textbf{(M1)} for substituting $a = 30$ into their regression equation.
  \textbf{(A1)} for $186$ (3 s.f.). Follow through from their $m$ and $c$.}

  \item[(d)] State whether this estimate is an example of interpolation or
  extrapolation. Give a reason for your answer. \hfill [1]

  Interpolation, because $30$ is within the range of the data
  ($18 \leq a \leq 45$). \hfill \textbf{(R1)}

  \textit{Award \textbf{(R1)} for ``interpolation'' with a correct reason referencing
  the data range. Do not award if the reason is missing or incorrect.}
\end{enumerate}

%--- Problem 2: Normal + Binomial + Conditional [8 marks] ---
\item[2.] [Maximum mark: 8]

The time, $T$ minutes, that customers wait in line at a coffee shop
is normally distributed with a mean of 4.5 minutes and a standard deviation
of 1.2 minutes.

\begin{enumerate}[itemsep=0.3cm]
  \item[(a)] Find $\mathrm{P}(T > 6)$. \hfill [2]

  \textit{GDC: normalcdf with lower $= 6$, upper $= \infty$, $\mu = 4.5$, $\sigma = 1.2$.}

  $\mathrm{P}(T > 6) = 1 - \mathrm{P}(T \leq 6)$ \hfill \textbf{(M1)}

  $= 0.106$ \hfill \textbf{(A1)}

  \textit{Award \textbf{(M1)} for a correct probability statement or GDC setup.
  \textbf{(A1)} for $0.106$ (3 s.f.).}

  \item[(b)] Find $\mathrm{P}(3 < T < 6)$. \hfill [2]

  \textit{GDC: normalcdf with lower $= 3$, upper $= 6$, $\mu = 4.5$, $\sigma = 1.2$.}

  $\mathrm{P}(3 < T < 6) = \mathrm{P}(T < 6) - \mathrm{P}(T < 3)$ \hfill \textbf{(M1)}

  $= 0.789$ \hfill \textbf{(A1)}

  \textit{Award \textbf{(M1)} for a correct approach (difference of two probabilities or direct GDC entry).
  \textbf{(A1)} for $0.789$ (3 s.f.).}

  \item[(c)] On a particular morning, 10 customers each wait in line independently.
  Find the probability that exactly 2 of them wait more than 6 minutes. \hfill [2]

  Let $X \sim \mathrm{B}(10,\, 0.1056\ldots)$ \hfill \textbf{(M1)}

  \textit{GDC: binompdf$(10,\, 0.1056,\, 2)$}

  $\mathrm{P}(X = 2) = 0.206$ \hfill \textbf{(A1)}

  \textit{Award \textbf{(M1)} for recognizing a binomial distribution with $n = 10$
  and $p$ from part (a). Award \textbf{(A1)} for a correct answer.
  Follow through from their answer to part (a).
  \textbf{(A1)} for $0.206$ (3 s.f.).}

  \item[(d)] A customer is selected at random. Given that they waited more than
  3 minutes, find the probability that they waited more than 6 minutes. \hfill [2]

  $\mathrm{P}(T > 6 \mid T > 3) = \dfrac{\mathrm{P}(T > 6)}{\mathrm{P}(T > 3)}$ \hfill \textbf{(M1)}

  $\mathrm{P}(T > 3) = 0.894$

  $= \dfrac{0.1056\ldots}{0.8944\ldots} = 0.118$ \hfill \textbf{(A1)}

  \textit{Award \textbf{(M1)} for correct use of the conditional probability formula.
  Award \textbf{(A1)} for a correct answer.
  \textbf{(A1)} for $0.118$ (3 s.f.).}
\end{enumerate}

\end{enumerate}
