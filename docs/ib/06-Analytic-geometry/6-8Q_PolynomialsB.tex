%--- Problem 1: Standard form to factored and vertex form [8 marks] ---
\item[1.] [Maximum mark: 8]

Let $f(x) = x^2 + 6x - 16$.

\begin{enumerate}[itemsep=0.15cm]
  \item Write $f(x)$ in factored form, such that $f(x) = (x - a)(x - b)$. \hfill [2]
  \item Write down the $x$-intercepts of the graph of $f$. \hfill [1]
  \item Write down the $y$-intercept. \hfill [1]
  \item Write $f(x)$ in vertex form, i.e. $f(x) = (x - h)^2 + k$. \hfill [3]
  \item Write down the coordinates of the vertex of the parabola. \hfill [1]
\end{enumerate}

%--- Problem 2: Vertex form conversions [7 marks] ---
\item[2.] [Maximum mark: 7]

Consider the function $g(x) = -(x - 5)^2 + 4$.

\begin{enumerate}[itemsep=0.15cm]
  \item Write down the coordinates of the vertex. \hfill [2]
  \item Write down the equation of the axis of symmetry. \hfill [1]
  \item Show that $g(x) = -x^2 + 10x - 21$. \hfill [2]
  \item Solve $g(x) = 0$ to find the $x$-intercepts. \hfill [2]
\end{enumerate}

%--- Problem 3: Intersection of line and parabola [6 marks] ---
\item[3.] [Maximum mark: 6]

The line $y = 4x$ and the parabola $y = x^2 - 5$ are drawn
on the same axes.

\begin{enumerate}[itemsep=0.15cm]
  \item Show that the $x$-coordinates of the points of intersection
  satisfy \\[0.25cm]
  $x^2 - 4x - 5 = 0$. \hfill [1]
  \item Show that $x^2 - 4x - 5=0$ has two distinct real roots. \hfill [1]
  \item Factorise $x^2 - 4x - 5$. \hfill [1]
  \item Find the $x$-coordinates of the points of intersection. \hfill [1]
  \item Find the coordinates of both points of intersection as $(x, y)$ pairs. \hfill [2]
\end{enumerate}

%--- Problem 4: Discriminant with parameter [7 marks] ---
\item[4.] [Maximum mark: 7]

The equation $2x^2 - kx + 32 = 0$ has two equal roots.

\begin{enumerate}[itemsep=0.15cm]
  \item Write down the discriminant in terms of $k$. \hfill [1]
  \item Find the values of $k$. \hfill [2]
  \item For the positive value of $k$, find the repeated root. \hfill [2]
  \item State the values of $k$ for which the equation has no
  real roots. \hfill [2]
\end{enumerate}
