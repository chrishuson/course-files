\subsubsection*{6.8 Quiz: Polynomials --- Markscheme (b)}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: f(x) = x^2 + 6x - 16 [8 marks] ---
\item[1.] [Maximum mark: 8]

Let $f(x) = x^2 + 6x - 16$.

\medskip

\textbf{(a)} Write $f(x)$ in factored form $(x - a)(x - b)$. \hfill [2]

$f(x) = (x - 2)(x - (-8)) = (x - 2)(x + 8)$ \hfill \textbf{A1A1}

\emph{Accept $(x + 8)(x - 2)$.}

\medskip

\textbf{(b)} Write down the $x$-intercepts. \hfill [1]

$x = 2$ and $x = -8$ \hfill \textbf{A1}

\emph{Accept $(2, 0)$ and $(-8, 0)$.}

\emph{FT: award A1 for $x$-intercepts read from the candidate's factors in (a).}

\medskip

\textbf{(c)} Write down the $y$-intercept. \hfill [1]

$f(0) = -16$, so the $y$-intercept is $(0, -16)$. \hfill \textbf{A1}

\emph{Accept $-16$.}

\emph{FT: award A1 for $y$-intercept correctly computed from the candidate's factored form in (a).}

\medskip

\textbf{(d)} Write $f(x)$ in vertex form $(x - h)^2 + k$. \hfill [3]

Axis of symmetry: $x = \dfrac{-8 + 2}{2} = -3$ \hfill \textbf{M1}

$f(x) = (x + 3)^2 - 25$ \hfill \textbf{A1A1}

\emph{Accept $h = -3$, $k = -25$.}

\emph{M1 may also be awarded for any attempt at completing the square.}

\medskip

\textbf{(e)} Write down the coordinates of the vertex. \hfill [1]

Vertex: $(-3, -25)$ \hfill \textbf{A1}

\emph{FT: award A1 for vertex coordinates read from the candidate's vertex form in (d).}

\bigskip
\hrule
\bigskip

%--- Problem 2: g(x) = -(x - 5)^2 + 4 [7 marks] ---
\item[2.] [Maximum mark: 7]

Consider the function $g(x) = -(x - 5)^2 + 4$.

\medskip

\textbf{(a)} Write down the coordinates of the vertex. \hfill [2]

Vertex: $(5, 4)$ \hfill \textbf{A1A1}

\medskip

\textbf{(b)} Write down the equation of the axis of symmetry. \hfill [1]

$x = 5$ \hfill \textbf{A1}

\medskip

\textbf{(c)} Show that $g(x) = -x^2 + 10x - 21$. \hfill [2]

$g(x) = -(x - 5)^2 + 4 = -(x^2 - 10x + 25) + 4$ \hfill \textbf{M1}

$= -x^2 + 10x - 25 + 4 = -x^2 + 10x - 21$ \hfill \textbf{A1 AG}

\medskip

\textbf{(d)} Solve $g(x) = 0$ to find the $x$-intercepts. \hfill [2]

$-x^2 + 10x - 21 = 0$

$x^2 - 10x + 21 = 0$

$(x - 3)(x - 7) = 0$

$x = 3$ or $x = 7$ \hfill \textbf{A1A1}

\newpage

%--- Problem 3: y = 4x and y = x^2 - 5 [6 marks] ---
\item[3.] [Maximum mark: 6]

The line $y = 4x$ and the parabola $y = x^2 - 5$.

\medskip

\textbf{(a)} Show that the $x$-coordinates of intersection
satisfy $x^2 - 4x - 5 = 0$. \hfill [1]

At intersection: $x^2 - 5 = 4x$ \hfill \textbf{M1}

$x^2 - 4x - 5 = 0$ \hfill \textbf{AG}

\medskip

\textbf{(b)} Show that $x^2 - 4x - 5 = 0$ has two distinct
real roots. \hfill [1]

$\Delta = (-4)^2 - 4(1)(-5) = 16 + 20 = 36 > 0$ \hfill \textbf{M1}

Since $\Delta > 0$, there are two distinct real roots.

\medskip

\textbf{(c)} Factorise $x^2 - 4x - 5$. \hfill [1]

$(x - 5)(x + 1)$ \hfill \textbf{A1}

\medskip

\textbf{(d)} Find the $x$-coordinates of the points
of intersection. \hfill [1]

$x = 5$ and $x = -1$ \hfill \textbf{A1}

\emph{FT: award A1 for $x$-coordinates read from the candidate's factors in (c).}

\medskip

\textbf{(e)} Find the coordinates of both points
of intersection. \hfill [2]

$x = 5$: $y = 4(5) = 20$, so $(5, 20)$ \hfill \textbf{A1}

$x = -1$: $y = 4(-1) = -4$, so $(-1, -4)$ \hfill \textbf{A1}

\emph{FT: award A1 for each $y$-value correctly computed from the candidate's $x$-value.}

\bigskip
\hrule
\bigskip

%--- Problem 4: 2x^2 - kx + 32 = 0 [7 marks] ---
\item[4.] [Maximum mark: 7]

The equation $2x^2 - kx + 32 = 0$ has two equal roots.

\medskip

\textbf{(a)} Write down the discriminant in terms of $k$. \hfill [1]

$\Delta = (-k)^2 - 4(2)(32) = k^2 - 256$ \hfill \textbf{M1}

\medskip

\textbf{(b)} Find the values of $k$. \hfill [2]

Equal roots: $\Delta = 0$ \hfill \textbf{M1}

$k^2 - 256 = 0$

$k^2 = 256$

$k = \pm 16$ \hfill \textbf{A1}

\medskip

\textbf{(c)} For $k = 16$, find the repeated root. \hfill [2]

$2x^2 - 16x + 32 = 0$

$x^2 - 8x + 16 = 0$ \hfill \textbf{M1}

$(x - 4)^2 = 0$

$x = 4$ \hfill \textbf{A1}

\medskip

\textbf{(d)} State the values of $k$ for which the equation
has no real roots. \hfill [2]

$\Delta < 0$ \hfill \textbf{M1}

$k^2 - 256 < 0$

$k^2 < 256$

$-16 < k < 16$ \hfill \textbf{A1}

\emph{Accept $|k| < 16$.}

\end{enumerate}
