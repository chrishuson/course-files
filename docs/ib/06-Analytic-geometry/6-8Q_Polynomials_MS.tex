\subsubsection*{6.8 Quiz: Polynomials --- Markscheme}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: f(x) = x^2 - 4x - 12 [8 marks] ---
\item[1.] [Maximum mark: 8]

Let $f(x) = x^2 - 4x - 12$.

\medskip

\textbf{(a)} Write $f(x)$ in factored form $(x - a)(x - b)$. \hfill [2]

$f(x) = (x - 6)(x - (-2)) = (x - 6)(x + 2)$ \hfill \textbf{A1A1}

\emph{Accept $(x + 2)(x - 6)$.}

\medskip

\textbf{(b)} Write down the $x$-intercepts. \hfill [1]

$x = 6$ and $x = -2$ \hfill \textbf{A1}

\emph{Accept $(6, 0)$ and $(-2, 0)$.}

\emph{FT: award A1 for $x$-intercepts read from the candidate's factors in (a).}

\medskip

\textbf{(c)} Write down the $y$-intercept. \hfill [1]

$f(0) = -12$, so the $y$-intercept is $(0, -12)$. \hfill \textbf{A1}

\emph{Accept $-12$.}

\emph{FT: award A1 for $y$-intercept correctly computed from the candidate's factored form in (a).}

\medskip

\textbf{(d)} Write $f(x)$ in vertex form $(x - h)^2 + k$. \hfill [3]

Axis of symmetry: $x = \dfrac{-2 + 6}{2} = 2$ \hfill \textbf{M1}

$f(x) = (x - 2)^2 - 16$ \hfill \textbf{A1A1}

\emph{Accept $h = 2$, $k = -16$.}

\emph{M1 may also be awarded for any attempt at completing the square.}

\medskip

\textbf{(e)} Write down the coordinates of the vertex. \hfill [1]

Vertex: $(2, -16)$ \hfill \textbf{A1}

\emph{FT: award A1 for vertex coordinates read from the candidate's vertex form in (d).}

\bigskip
\hrule
\bigskip

%--- Problem 2: g(x) = -(x - 3)^2 + 16 [7 marks] ---
\item[2.] [Maximum mark: 7]

Consider the function $g(x) = -(x - 3)^2 + 16$.

\medskip

\textbf{(a)} Write down the coordinates of the vertex. \hfill [2]

Vertex: $(3, 16)$ \hfill \textbf{A1A1}

\medskip

\textbf{(b)} Write down the equation of the axis of symmetry. \hfill [1]

$x = 3$ \hfill \textbf{A1}

\medskip

\textbf{(c)} Show that $g(x) = -x^2 + 6x + 7$. \hfill [2]

$g(x) = -(x - 3)^2 + 16 = -(x^2 - 6x + 9) + 16$ \hfill \textbf{M1}

$= -x^2 + 6x - 9 + 16 = -x^2 + 6x + 7$ \hfill \textbf{A1 AG}

\medskip

\textbf{(d)} Solve $g(x) = 0$ to find the $x$-intercepts. \hfill [2]

$-x^2 + 6x + 7 = 0$

$x^2 - 6x - 7 = 0$

$(x - 7)(x + 1) = 0$

$x = 7$ or $x = -1$ \hfill \textbf{A1A1}

\newpage

%--- Problem 3: y = 3x and y = x^2 - 4 [6 marks] ---
\item[3.] [Maximum mark: 6]

The line $y = 3x$ and the parabola $y = x^2 - 4$.

\medskip

\textbf{(a)} Show that the $x$-coordinates of intersection
satisfy $x^2 - 3x - 4 = 0$. \hfill [1]

At intersection: $x^2 - 4 = 3x$ \hfill \textbf{M1}

$x^2 - 3x - 4 = 0$ \hfill \textbf{AG}

\medskip

\textbf{(b)} Show that $x^2 - 3x - 4 = 0$ has two distinct
real roots. \hfill [1]

$\Delta = (-3)^2 - 4(1)(-4) = 9 + 16 = 25 > 0$ \hfill \textbf{M1}

Since $\Delta > 0$, there are two distinct real roots.

\medskip

\textbf{(c)} Factorise $x^2 - 3x - 4$. \hfill [1]

$(x - 4)(x + 1)$ \hfill \textbf{A1}

\medskip

\textbf{(d)} Find the $x$-coordinates of the points
of intersection. \hfill [1]

$x = 4$ and $x = -1$ \hfill \textbf{A1}

\emph{FT: award A1 for $x$-coordinates read from the candidate's factors in (c).}

\medskip

\textbf{(e)} Find the coordinates of both points
of intersection. \hfill [2]

$x = 4$: $y = 3(4) = 12$, so $(4, 12)$ \hfill \textbf{A1}

$x = -1$: $y = 3(-1) = -3$, so $(-1, -3)$ \hfill \textbf{A1}

\emph{FT: award A1 for each $y$-value correctly computed from the candidate's $x$-value.}

\bigskip
\hrule
\bigskip

%--- Problem 4: 3x^2 - kx + 12 = 0 [7 marks] ---
\item[4.] [Maximum mark: 7]

The equation $3x^2 - kx + 12 = 0$ has two equal roots.

\medskip

\textbf{(a)} Write down the discriminant in terms of $k$. \hfill [1]

$\Delta = (-k)^2 - 4(3)(12) = k^2 - 144$ \hfill \textbf{M1}

\medskip

\textbf{(b)} Find the values of $k$. \hfill [2]

Equal roots: $\Delta = 0$ \hfill \textbf{M1}

$k^2 - 144 = 0$

$k^2 = 144$

$k = \pm 12$ \hfill \textbf{A1}

\medskip

\textbf{(c)} For $k = 12$, find the repeated root. \hfill [2]

$3x^2 - 12x + 12 = 0$

$x^2 - 4x + 4 = 0$ \hfill \textbf{M1}

$(x - 2)^2 = 0$

$x = 2$ \hfill \textbf{A1}

\medskip

\textbf{(d)} State the values of $k$ for which the equation
has no real roots. \hfill [2]

$\Delta < 0$ \hfill \textbf{M1}

$k^2 - 144 < 0$

$k^2 < 144$

$-12 < k < 12$ \hfill \textbf{A1}

\emph{Accept $|k| < 12$.}

\end{enumerate}
