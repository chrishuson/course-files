\subsubsection*{8.5 Quiz: Rational Functions and Transformations --- Markscheme}

\medskip

%================================================================

%================================================================
\noindent\textbf{\large Version (a)}

\bigskip

%--- Problem 1 (a) [8 marks]
\noindent\textbf{1.} $\displaystyle f(x)=\dfrac{2}{x-3}+1$.

\medskip

\textbf{(a)} Vertical asymptote: $x=3$. \hfill \textbf{A1}

Horizontal asymptote: $y=1$. \hfill \textbf{A1}

\medskip

\textbf{(b)} \emph{``Describe fully'' requires words, not symbols alone. One mark
per correctly described transformation (A1A1A1).}

A vertical stretch (stretch / dilate / scale), scale factor $2$ --- direction (vertical) stated. \hfill \textbf{A1}

A translation (translate / shift / slide) $3$ to the right --- ``right'' explicit. \hfill \textbf{A1}

A translation $1$ up (vertical direction may be omitted; assumed upward). \hfill \textbf{A1}

\emph{Accept ``factor of two''/``doubled vertically''; ``parallel to the $y$-axis''
for the stretch direction; ``positive $x$-direction'' for right. The word
``units'' is optional. \textbf{Reject ``move''.}}

\emph{Order: the stretch must precede the vertical translation; the horizontal
translation may go in any position (deduct $1$ mark if the vertical translation
precedes the stretch).}

\emph{Accept the two translations as a vector
$\left(\begin{smallmatrix}3\\1\end{smallmatrix}\right)$ for both translation marks,
provided a translation word is used. A purely symbolic description with no
transformation words (e.g.\ $\tfrac1x\to\tfrac2x\to\tfrac{2}{x-3}\to\tfrac{2}{x-3}+1$)
scores at most $1$ of $3$.}

\medskip

\textbf{(c)} $f(0)=\dfrac{2}{-3}+1=\dfrac{1}{3}$, \ so $\left(0,\tfrac{1}{3}\right)$. \hfill \textbf{A1}

\emph{Accept $y=\tfrac{1}{3}$.}

\medskip

\textbf{(d)} $\dfrac{2}{x-3}+1=0$, \ \ $\dfrac{2}{x-3}=-1$, \ \ $x-3=-2$ \hfill \textbf{M1}

$x=1$, \ so $(1,0)$. \hfill \textbf{A1}

\emph{Accept $x=1$.}

\medskip

%--- Problem 2 (a) [8 marks]
\noindent\textbf{2.} $\displaystyle g(x)=\dfrac{2x+1}{x-1}$, \ $x\ne 1$.

\medskip

\textbf{(a)} Vertical asymptote: $x=1$. \hfill \textbf{A1}

Horizontal asymptote: $y=2$ \ (ratio of leading coefficients $=\tfrac{2}{1}$). \hfill \textbf{A1}

\medskip

\textbf{(b)} $x$-intercept: $2x+1=0 \Rightarrow x=-\tfrac{1}{2}$, \ so $\left(-\tfrac{1}{2},0\right)$. \hfill \textbf{A1}

$y$-intercept: $g(0)=\dfrac{1}{-1}=-1$, \ so $(0,-1)$. \hfill \textbf{A1}

\emph{Accept $x=-\tfrac{1}{2}$ and $y=-1$.}

\medskip

\textbf{(c)} Let $y=\dfrac{2x+1}{x-1}$ and swap $x$ and $y$: \ \ $x=\dfrac{2y+1}{y-1}$ \hfill \textbf{M1}

$x(y-1)=2y+1$, \ \ $xy-2y=x+1$, \ \ $y(x-2)=x+1$ \hfill \textbf{A1}

$g^{-1}(x)=\dfrac{x+1}{x-2}$ \hfill \textbf{A1}

\medskip

\textbf{(d)} Domain of $g^{-1}$: $x\in\mathbb{R}$, $x\ne 2$. \hfill \textbf{A1}

\emph{FT: award A1 for $x\ne$ (the candidate's horizontal-asymptote value of $g$ from (a)).}

\bigskip\hrule\bigskip

\noindent\textbf{\large Version (b)}

\bigskip

%--- Problem 1 (b) [8 marks]
\noindent\textbf{1.} $\displaystyle f(x)=\dfrac{3}{x+2}-1$.

\medskip

\textbf{(a)} Vertical asymptote: $x=-2$. \hfill \textbf{A1}

Horizontal asymptote: $y=-1$. \hfill \textbf{A1}

\medskip

\textbf{(b)} \emph{``Describe fully'' requires words, not symbols alone. One mark
per correctly described transformation (A1A1A1).}

A vertical stretch (stretch / dilate / scale), scale factor $3$ --- direction (vertical) stated. \hfill \textbf{A1}

A translation (translate / shift / slide) $2$ to the left --- ``left'' explicit. \hfill \textbf{A1}

A translation $1$ \textbf{down} --- ``down'' must be explicit (the upward default does not apply here). \hfill \textbf{A1}

\emph{Accept ``factor of three''/``tripled vertically''; ``parallel to the $y$-axis''
for the stretch direction; ``negative $x$-direction'' for left. The word
``units'' is optional. \textbf{Reject ``move''.}}

\emph{Order: the stretch must precede the vertical translation; the horizontal
translation may go in any position (deduct $1$ mark if the vertical translation
precedes the stretch).}

\emph{Accept the two translations as a vector
$\left(\begin{smallmatrix}-2\\-1\end{smallmatrix}\right)$ for both translation marks,
provided a translation word is used. A purely symbolic description with no
transformation words (e.g.\ $\tfrac1x\to\tfrac3x\to\tfrac{3}{x+2}\to\tfrac{3}{x+2}-1$)
scores at most $1$ of $3$.}

\medskip

\textbf{(c)} $f(0)=\dfrac{3}{2}-1=\dfrac{1}{2}$, \ so $\left(0,\tfrac{1}{2}\right)$. \hfill \textbf{A1}

\emph{Accept $y=\tfrac{1}{2}$.}

\medskip

\textbf{(d)} $\dfrac{3}{x+2}-1=0$, \ \ $\dfrac{3}{x+2}=1$, \ \ $x+2=3$ \hfill \textbf{M1}

$x=1$, \ so $(1,0)$. \hfill \textbf{A1}

\emph{Accept $x=1$.}

\medskip

%--- Problem 2 (b) [8 marks]
\noindent\textbf{2.} $\displaystyle g(x)=\dfrac{3x-2}{x+1}$, \ $x\ne -1$.

\medskip

\textbf{(a)} Vertical asymptote: $x=-1$. \hfill \textbf{A1}

Horizontal asymptote: $y=3$ \ (ratio of leading coefficients $=\tfrac{3}{1}$). \hfill \textbf{A1}

\medskip

\textbf{(b)} $x$-intercept: $3x-2=0 \Rightarrow x=\tfrac{2}{3}$, \ so $\left(\tfrac{2}{3},0\right)$. \hfill \textbf{A1}

$y$-intercept: $g(0)=\dfrac{-2}{1}=-2$, \ so $(0,-2)$. \hfill \textbf{A1}

\emph{Accept $x=\tfrac{2}{3}$ and $y=-2$.}

\medskip

\textbf{(c)} Let $y=\dfrac{3x-2}{x+1}$ and swap $x$ and $y$: \ \ $x=\dfrac{3y-2}{y+1}$ \hfill \textbf{M1}

$x(y+1)=3y-2$, \ \ $xy-3y=-x-2$, \ \ $y(x-3)=-(x+2)$ \hfill \textbf{A1}

$g^{-1}(x)=-\dfrac{x+2}{x-3}$ \hfill \textbf{A1}

\emph{Accept the equivalent form $\dfrac{x+2}{3-x}$.}

\medskip

\textbf{(d)} Domain of $g^{-1}$: $x\in\mathbb{R}$, $x\ne 3$. \hfill \textbf{A1}

\emph{FT: award A1 for $x\ne$ (the candidate's horizontal-asymptote value of $g$ from (a)).}

\noindent\textbf{\large Version (c)}

\bigskip

%--- Problem 1 (c) [8 marks]
\noindent\textbf{1.} $\displaystyle f(x)=\dfrac{4}{x-1}+2$.

\medskip

\textbf{(a)} Vertical asymptote: $x=1$. \hfill \textbf{A1}

Horizontal asymptote: $y=2$. \hfill \textbf{A1}

\medskip

\textbf{(b)} \emph{``Describe fully'' requires words, not symbols alone. One mark
per correctly described transformation (A1A1A1).}

A vertical stretch (stretch / dilate / scale), scale factor $4$ --- direction (vertical) stated. \hfill \textbf{A1}

A translation (translate / shift / slide) $1$ to the right --- ``right'' explicit. \hfill \textbf{A1}

A translation $2$ up (vertical direction may be omitted; assumed upward). \hfill \textbf{A1}

\emph{Accept ``factor of four''/``quadrupled vertically''; ``parallel to the $y$-axis''
for the stretch direction; ``positive $x$-direction'' for right. The word
``units'' is optional. \textbf{Reject ``move''.}}

\emph{Order: the stretch must precede the vertical translation; the horizontal
translation may go in any position (deduct $1$ mark if the vertical translation
precedes the stretch).}

\emph{Accept the two translations as a vector
$\left(\begin{smallmatrix}1\\2\end{smallmatrix}\right)$ for both translation marks,
provided a translation word is used. A purely symbolic description with no
transformation words (e.g.\ $\tfrac1x\to\tfrac4x\to\tfrac{4}{x-1}\to\tfrac{4}{x-1}+2$)
scores at most $1$ of $3$.}

\medskip

\textbf{(c)} $f(0)=\dfrac{4}{-1}+2=-2$, \ so $(0,-2)$. \hfill \textbf{A1}

\emph{Accept $y=-2$.}

\medskip

\textbf{(d)} $\dfrac{4}{x-1}+2=0$, \ \ $\dfrac{4}{x-1}=-2$, \ \ $x-1=-2$ \hfill \textbf{M1}

$x=-1$, \ so $(-1,0)$. \hfill \textbf{A1}

\emph{Accept $x=-1$.}

\medskip

%--- Problem 2 (c) [8 marks]
\noindent\textbf{2.} $\displaystyle g(x)=\dfrac{4x+3}{x-2}$, \ $x\ne 2$.

\medskip

\textbf{(a)} Vertical asymptote: $x=2$. \hfill \textbf{A1}

Horizontal asymptote: $y=4$ \ (ratio of leading coefficients $=\tfrac{4}{1}$). \hfill \textbf{A1}

\medskip

\textbf{(b)} $x$-intercept: $4x+3=0 \Rightarrow x=-\tfrac{3}{4}$, \ so $\left(-\tfrac{3}{4},0\right)$. \hfill \textbf{A1}

$y$-intercept: $g(0)=\dfrac{3}{-2}=-\tfrac{3}{2}$, \ so $\left(0,-\tfrac{3}{2}\right)$. \hfill \textbf{A1}

\emph{Accept $x=-\tfrac{3}{4}$ and $y=-\tfrac{3}{2}$.}

\medskip

\textbf{(c)} Let $y=\dfrac{4x+3}{x-2}$ and swap $x$ and $y$: \ \ $x=\dfrac{4y+3}{y-2}$ \hfill \textbf{M1}

$x(y-2)=4y+3$, \ \ $xy-4y=2x+3$, \ \ $y(x-4)=2x+3$ \hfill \textbf{A1}

$g^{-1}(x)=\dfrac{2x+3}{x-4}$ \hfill \textbf{A1}

\medskip

\textbf{(d)} Domain of $g^{-1}$: $x\in\mathbb{R}$, $x\ne 4$. \hfill \textbf{A1}

\emph{FT: award A1 for $x\ne$ (the candidate's horizontal-asymptote value of $g$ from (a)).}
