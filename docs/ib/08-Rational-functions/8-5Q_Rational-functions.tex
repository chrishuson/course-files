\subsubsection*{8.5 Quiz: Rational Functions and Transformations (no calculator)}

\begin{enumerate}[itemsep=0.15cm]

%--- Problem 1: translated reciprocal — asymptotes, transformations, intercepts [8 marks] ---
%--- Source: informed by 8-1CW (asymptotes/intercepts) and 8-3CW (transformations of 1/x) ---
\item[1.] [Maximum mark: 8]

Consider the function $f(x) = \dfrac{2}{x - 3} + 1$.

\begin{enumerate}[itemsep=0.15cm]
  \item Write down the equations of the vertical and horizontal asymptotes of the graph of $f$. \hfill [2]
  \item Describe fully the sequence of transformations that maps the graph of $y = \dfrac{1}{x}$ onto the graph of $f$. \hfill [3]
  \item Find the $y$-intercept of the graph of $f$. \hfill [1]
  \item Find the $x$-intercept of the graph of $f$. \hfill [2]
\end{enumerate}

%--- Source: informed by 8-4CW Challenge Q1 (qb_id 22M.2.SL.TZ1.8 family) ---
\item[2.] [Maximum mark: 8]

The function $g$ is defined by $g(x) = \dfrac{2x + 1}{x - 1}$, \ $x \in \mathbb{R}, \ x \ne 1$.

\begin{enumerate}[itemsep=0.15cm]
  \item Write down the equations of the vertical and horizontal asymptotes of the graph of $g$. \hfill [2]
  \item Find the coordinates of the $x$-intercept and of the $y$-intercept of the graph of $g$. \hfill [2]
  \item Find $g^{-1}(x)$. \hfill [3]
  \item State the domain of $g^{-1}$. \hfill [1]
\end{enumerate}

\end{enumerate}
