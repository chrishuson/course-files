\subsubsection*{8.7 Classwork: Function Composition}

\begin{enumerate}[itemsep=0.15cm]

%--- Problem 1: composition and inverse from a table [6 marks] ---
\item[1.]

The functions $f$ and $g$ are defined in the following table.

\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & $-2$ & $0$ & $1$ & $3$ & $5$ \\ \hline
$f(x)$ & $3$ & $1$ & $5$ & $-2$ & $0$ \\ \hline
$g(x)$ & $1$ & $-2$ & $0$ & $5$ & $3$ \\ \hline
\end{tabular}
\end{center}

\begin{enumerate}[itemsep=0.15cm]
  \item Write down the value of $f(1)$. \vfill
  \item Find the value of $(g \circ f)(1)$. \vfill
  \item Find the value of $(f \circ g)(-2)$. \vfill
  \item Find the value of $f^{-1}(-2)$. \vfill
\end{enumerate}

\newpage

%--- Problem 2: algebraic composition with a simple quadratic [7 marks] ---
\item[2.]

Let $f(x) = 2x - 1$ \ and \ $g(x) = x^2 + 3$.

\begin{enumerate}[itemsep=0.15cm]
  \item Find $(f \circ g)(x)$. \vfill
  \item Find $(g \circ f)(x)$, giving your answer in expanded form. \vfill
  \item Evaluate $(g \circ f)(-1)$. \vfill
  \item Solve the equation $(f \circ g)(x) = 13$. \vfill
\end{enumerate}

\newpage

%--- Problem 3: composition of a rational function, with domain [7 marks] ---
\item[3.]

Let $f(x) = \dfrac{1}{x - 2}$ \ and \ $g(x) = x + 5$.

\begin{enumerate}[itemsep=0.15cm]
  \item Find $(f \circ g)(x)$, simplifying your answer. \vfill
  \item Write down the value of $x$ that must be excluded from the domain of $f \circ g$. \vfill
  \item Find $(g \circ f)(x)$, giving your answer as a single fraction. \vfill
  \item Find $(f \circ f)(x)$, simplifying your answer. \vfill
\end{enumerate}

\newpage

%--- Problem 4: recover the inner function, then invert [6 marks] ---
\item[4.]

The function $f$ is defined by $f(x) = 3x - 4$, \ and $(f \circ g)(x) = 6x + 2$.

\begin{enumerate}[itemsep=0.15cm]
  \item Find an expression for $g(x)$. \vfill
  \item Find $g^{-1}(x)$. \vfill
\end{enumerate}

\end{enumerate}
