\subsubsection*{8.7 Classwork: Function Composition --- Markscheme}

\medskip

%--- Problem 1 [6 marks]
\noindent\textbf{1.} Functions $f$ and $g$ given by the table.

\medskip

\textbf{(a)} $f(1)=5$. \hfill \textbf{A1}

\medskip

\textbf{(b)} $(g \circ f)(1)=g\big(f(1)\big)=g(5)$ \hfill \textbf{M1}

$=3$. \hfill \textbf{A1}

\medskip

\textbf{(c)} $(f \circ g)(-2)=f\big(g(-2)\big)=f(1)$ \hfill \textbf{M1}

$=5$. \hfill \textbf{A1}

\medskip

\textbf{(d)} $f^{-1}(-2)=3$ \ (since $f(3)=-2$). \hfill \textbf{A1}

\bigskip\hrule\bigskip

%--- Problem 2 [7 marks]
\noindent\textbf{2.} $f(x)=2x-1$, \ $g(x)=x^2+3$.

\medskip

\textbf{(a)} $(f \circ g)(x)=2\big(x^2+3\big)-1$ \hfill \textbf{M1}

$=2x^2+5$. \hfill \textbf{A1}

\medskip

\textbf{(b)} $(g \circ f)(x)=(2x-1)^2+3=4x^2-4x+1+3$ \hfill \textbf{M1}

$=4x^2-4x+4$. \hfill \textbf{A1}

\medskip

\textbf{(c)} $(g \circ f)(-1)=g(-3)=9+3=12$. \hfill \textbf{A1}

\emph{Or $4(-1)^2-4(-1)+4=12$.}

\medskip

\textbf{(d)} $2x^2+5=13$, \ \ $2x^2=8$, \ \ $x^2=4$ \hfill \textbf{M1}

$x=\pm 2$. \hfill \textbf{A1}

\emph{Award A0 if only $x=2$ is given.}

\bigskip\hrule\bigskip

%--- Problem 3 [7 marks]
\noindent\textbf{3.} $f(x)=\dfrac{1}{x-2}$, \ $g(x)=x+5$.

\medskip

\textbf{(a)} $(f \circ g)(x)=\dfrac{1}{(x+5)-2}$ \hfill \textbf{M1}

$=\dfrac{1}{x+3}$. \hfill \textbf{A1}

\medskip

\textbf{(b)} $x=-3$. \hfill \textbf{A1}

\medskip

\textbf{(c)} $(g \circ f)(x)=\dfrac{1}{x-2}+5=\dfrac{1+5(x-2)}{x-2}$ \hfill \textbf{M1}

$=\dfrac{5x-9}{x-2}$. \hfill \textbf{A1}

\medskip

\textbf{(d)} $(f \circ f)(x)=\dfrac{1}{\dfrac{1}{x-2}-2}=\dfrac{x-2}{1-2(x-2)}$ \hfill \textbf{M1}

$=\dfrac{x-2}{5-2x}$. \hfill \textbf{A1}

\bigskip\hrule\bigskip

%--- Problem 4 [6 marks]
\noindent\textbf{4.} $f(x)=3x-4$, \ $(f \circ g)(x)=6x+2$.

\medskip

\textbf{(a)} $f\big(g(x)\big)=3\,g(x)-4=6x+2$ \hfill \textbf{M1}

$3\,g(x)=6x+6$ \hfill \textbf{A1}

$g(x)=2x+2$. \hfill \textbf{A1}

\medskip

\textbf{(b)} Let $y=2x+2$ and swap $x$ and $y$: \ \ $x=2y+2$ \hfill \textbf{M1}

$y=\dfrac{x-2}{2}$ \hfill \textbf{A1}

$g^{-1}(x)=\dfrac{x-2}{2}$. \hfill \textbf{A1}
