\subsubsection*{4.19 Quiz: Normal and Binomial Distribution --- Markscheme}

\begin{enumerate}[itemsep=0.15cm]

%--- Problem 1: Simple Binomial, X ~ B(10, 0.20) [7 marks] ---
\item A market stall sells bags of apples. It is known that 20\% of the apples
      have bruises. A bag contains 10 randomly selected apples.
      Let $X$ be the number of bruised apples in the bag.

\begin{enumerate}[itemsep=0.3cm]
  \item Find the expected number of bruised apples in a bag, $E(X)$. \hfill [1]

  $E(X) = np = 10 \times 0.20 = 2$ \hfill \textbf{(A1)}

  \item Find the probability that none of the apples are bruised. \hfill [2]

  $\mathrm{P}(X = 0) = \binom{10}{0}(0.20)^{0}(0.80)^{10}$ \hfill \textbf{(M1)}

  $= 0.80^{10} = 0.107$ \hfill \textbf{(A1)}

  \item Find $\mathrm{P}(X = 3)$. \hfill [2]

  $\mathrm{P}(X = 3) = \binom{10}{3}(0.20)^{3}(0.80)^{7}$ \hfill \textbf{(M1)}

  $= 120 \times 0.008 \times 0.2097\ldots = 0.201$ \hfill \textbf{(A1)}

  \emph{GDC: binomPdf$(10,\, 0.20,\, 3) = 0.20133\ldots$}

  \item Find $\mathrm{P}(X \leq 3)$. \hfill [2]

  $\mathrm{P}(X \leq 3) = \mathrm{P}(X=0) + \mathrm{P}(X=1) + \mathrm{P}(X=2) + \mathrm{P}(X=3)$ \hfill \textbf{(M1)}

  $= 0.8791$ \hfill \textbf{(A1)}

  \emph{GDC: binomCdf$(10,\, 0.20,\, 0,\, 3) = 0.87913\ldots$}
\end{enumerate}

%--- Problem 2: Simple Normal, M ~ N(72, 8²) [7 marks] ---
\item The marks on a mathematics test follow a normal distribution
      with mean $\mu = 72$ and standard deviation $\sigma = 8$.

\begin{enumerate}[itemsep=0.3cm]
  \item Find the probability that a randomly selected student scores
        higher than 85. \hfill [2]

  $\mathrm{P}(M > 85)$ \hfill \textbf{(M1)}

  \emph{GDC: normCdf$(85,\, 10^{99},\, 72,\, 8) = 0.05208\ldots$}

  $\approx 0.0521$ \hfill \textbf{(A1)}

  \item Find the probability that a student's mark is between
        60 and 80. \hfill [2]

  $\mathrm{P}(60 < M < 80)$ \hfill \textbf{(M1)}

  \emph{GDC: normCdf$(60,\, 80,\, 72,\, 8) = 0.77453\ldots$}

  $\approx 0.775$ \hfill \textbf{(A1)}

  \item 75\% of the students score less than $h$ marks. Find $h$. \hfill [3]

  $\mathrm{P}(M < h) = 0.75$ \hfill \textbf{(M1)}

  \emph{GDC: invNorm$(0.75,\, 72,\, 8)$} \hfill \textbf{(M1)}

  $h = 77.396\ldots \approx 77.4$ \hfill \textbf{(A1)}
\end{enumerate}

%--- Problem 3: Combined Normal + Binomial, T ~ N(15, σ²) [11 marks] ---
\item The waiting time $T$ (minutes) at a bus stop follows a normal
      distribution with mean 15 minutes and standard deviation $\sigma$ minutes.
      It is known that 4\% of waits are longer than 20 minutes.

\begin{enumerate}[itemsep=0.3cm]
  \item Find the value of $\sigma$. \hfill [3]

  $\mathrm{P}(T > 20) = 0.04$, so $\mathrm{P}(T < 20) = 0.96$ \hfill \textbf{(M1)}

  \emph{GDC: invNorm$(0.96) = 1.7507\ldots$}

  $\dfrac{20 - 15}{\sigma} = 1.7507$ \hfill \textbf{(M1)}

  $\sigma = \dfrac{5}{1.7507} = 2.856\ldots \approx 2.86$ \hfill \textbf{(A1)}

  \item Find $\mathrm{P}(T < 12)$. \hfill [2]

  \emph{GDC: normCdf$(-10^{99},\, 12,\, 15,\, 2.856)$} \hfill \textbf{(M1)}

  $= 0.1468\ldots \approx 0.147$ \hfill \textbf{(A1)}

  \item On a particular day, 10 people wait at this bus stop. Find the probability
        that at most 2 of them wait more than 20 minutes. \hfill [3]

  $\mathrm{P}(T > 20) = 0.04$, so $Y \sim \mathrm{B}(10,\, 0.04)$ \hfill \textbf{(M1)}

  $\mathrm{P}(Y \leq 2)$ \hfill \textbf{(M1)}

  \emph{GDC: binomCdf$(10,\, 0.04,\, 0,\, 2) = 0.9938\ldots$}

  $\approx 0.994$ \hfill \textbf{(A1)}

  \item A person is selected at random. Given that they waited less than
        15 minutes, find the probability that they waited less than
        10 minutes. \hfill [3]

  $\mathrm{P}(T < 10 \mid T < 15)
    = \dfrac{\mathrm{P}(T < 10)}{\mathrm{P}(T < 15)}$ \hfill \textbf{(M1)}

  $\mathrm{P}(T < 15) = 0.5$ (since $\mu = 15$) \hfill \textbf{(A1)}

  \emph{GDC: normCdf$(-10^{99},\, 10,\, 15,\, 2.856) = 0.04006\ldots$}

  $\mathrm{P}(T < 10 \mid T < 15) = \dfrac{0.04006}{0.5} = 0.0801\ldots \approx 0.0801$ \hfill \textbf{(A1)}
\end{enumerate}

\end{enumerate}
