%--- Problem 3: Combined events with formulas ---
\item Events $A$ and $B$ are such that $\mathrm{P}(A) = 0.6$, $\mathrm{P}(B) = 0.5$, and
$\mathrm{P}(A \cap B) = 0.2$.
\begin{enumerate}[itemsep=3cm]
    \item Find $\mathrm{P}(A \cup B)$. (Show substitution into a formula.)
    \item Find $\mathrm{P}(A' \cap B')$.
    \item Find $\mathrm{P}(A \mid B)$. (Show substitution into a formula.)
    \item Determine whether $A$ and $B$ are independent. Justify your answer.
\end{enumerate}

        \hline
        Total    & & & \\
    \end{tabular}
    \end{center}
    \item Find $\mathrm{P}(A \cup B)$.
    \item Find $\mathrm{P}(B \mid A')$.

%--- Problem 5: Mutually exclusive ---
\item Events $A$ and $B$ are mutually exclusive, with $\mathrm{P}(A) = 0.35$ and $\mathrm{P}(B) = 0.25$.

    \item Find $\mathrm{P}(A \cup B)$.
    \item Find $\mathrm{P}(A' \cap B')$.
\end{enumerate} \vspace{0.5cm}
