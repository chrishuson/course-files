\subsubsection*{2.2 Classwork: Radicals and Exponents (no calculator) --- Markscheme}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: Laws of exponents (product, quotient, power, zero, negative, fractional) [14 marks] ---
\item[1.] [Maximum mark: 14]

\textbf{(a)} $3^{4} \times 3^{2} = 3^{4 + 2} = 3^{6}$ \hfill \textbf{A1}

\emph{Accept $729$.}

\textbf{(b)} $\frac{x^{7}}{x^{3}} = x^{7 - 3} = x^{4}$ \hfill \textbf{A1}

\textbf{(c)} $\left(a^{2}\right)^{3} = a^{2 \times 3} = a^{6}$ \hfill \textbf{A1}

\textbf{(d)} $\left(2x^{3}\right)^{2} = 2^{2} x^{3 \times 2} = 4x^{6}$ \hfill \textbf{A1A1}

\emph{A1 for the coefficient $4$, A1 for $x^{6}$. The common error is
$2x^{6}$: A0A1.}

\textbf{(e)} $\frac{6x^{5}}{2x^{2}} = \frac{6}{2} x^{5 - 2} = 3x^{3}$ \hfill \textbf{A1A1}

\emph{A1 for the coefficient $3$, A1 for $x^{3}$.}

\textbf{(f)} $7^{0} = 1$ \hfill \textbf{A1}

\textbf{(g)} $2^{-3} = \frac{1}{2^{3}} = \frac{1}{8}$ \hfill \textbf{A1}

\emph{Accept $0.125$. The common error is $-8$ or $-6$: A0.}

\textbf{(h)} $9^{\frac{1}{2}} = \sqrt{9} = 3$ \hfill \textbf{A1}

\textbf{(i)} $8^{\frac{2}{3}} = \left(\sqrt[3]{8}\right)^{2} = 2^{2}$ \hfill \textbf{M1}

$= 4$ \hfill \textbf{A1}

\emph{Accept $\sqrt[3]{8^{2}} = \sqrt[3]{64}$ for the M1.}

\textbf{(j)} $4^{-\frac{1}{2}} = \frac{1}{4^{\frac{1}{2}}} = \frac{1}{\sqrt{4}}$ \hfill \textbf{M1}

$= \frac{1}{2}$ \hfill \textbf{A1}

\emph{Accept $0.5$. Each one-mark part may be answered with the value alone.
In (i) and (j) a correct value with no working earns A1 only.}

\newpage

%--- Problem 2: Simplify radicals; add/subtract like radicals; rationalize sqrt denominator; conjugate product and conjugate rationalization [15 marks] ---
\item[2.] [Maximum mark: 15]

\textbf{(a)} $\sqrt{12} = \sqrt{4}\sqrt{3}$ \hfill \textbf{M1}

$= 2\sqrt{3}$ \hfill \textbf{A1}

\textbf{(b)} $\sqrt{50} = \sqrt{25}\sqrt{2}$ \hfill \textbf{M1}

$= 5\sqrt{2}$ \hfill \textbf{A1}

\emph{In (a) and (b), M1 for writing the number as a perfect square times
another factor. A correct answer with no working earns A1 only.}

\textbf{(c)} $\sqrt{2} \times \sqrt{8} = \sqrt{16} = 4$ \hfill \textbf{A1}

\textbf{(d)} $3\sqrt{5} + 4\sqrt{5} = 7\sqrt{5}$ \hfill \textbf{A1}

\emph{The common error is $7\sqrt{10}$: A0.}

\textbf{(e)} $\sqrt{18} - \sqrt{2} = 3\sqrt{2} - \sqrt{2}$ \hfill \textbf{M1}

$= 2\sqrt{2}$ \hfill \textbf{A1}

\emph{The common error is $\sqrt{16} = 4$: M0A0.}

\textbf{(f)} $\frac{6}{\sqrt{3}} \times \frac{\sqrt{3}}{\sqrt{3}}$ \hfill \textbf{M1}

$= \frac{6\sqrt{3}}{3}$

$= 2\sqrt{3}$ \hfill \textbf{A1}

\emph{Accept $\frac{6\sqrt{3}}{3}$ for the A1.}

\textbf{(g)} $\left(3 - \sqrt{5}\right)\left(3 + \sqrt{5}\right) = 9 + 3\sqrt{5} - 3\sqrt{5} - 5$ \hfill \textbf{M1}

$= 4$ \hfill \textbf{A1}

\emph{Accept the difference of two squares, $3^{2} - \left(\sqrt{5}\right)^{2}$,
for the M1.}

\textbf{(h)} $\frac{8}{3 - \sqrt{5}} \times \frac{3 + \sqrt{5}}{3 + \sqrt{5}}$ \hfill \textbf{M1}

$= \frac{8\left(3 + \sqrt{5}\right)}{4}$ \hfill \textbf{A1}

$= 2\left(3 + \sqrt{5}\right)$

$= 6 + 2\sqrt{5}$ \hfill \textbf{A1}

\emph{M1 for multiplying numerator and denominator by the conjugate
$3 + \sqrt{5}$. A1 for the denominator $4$, FT from (g). Accept
$2\left(3 + \sqrt{5}\right)$ for the final A1.}

\end{enumerate}
