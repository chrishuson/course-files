%--- Problem 1: Two 2x2 systems by elimination, scaling both equations; first one scaffolded step by step [7 marks] ---
\item[1.] [Maximum mark: 7]

\emph{Elimination is required. Substitution in place of elimination earns
M0, then A1A1 for correct values. FT the second value from the first.}

\textbf{(a)} $4x + 3y = 5$ \ (1), \quad $6x - 5y = 17$ \ (2)

(i) $3 \times (1)$: $12x + 9y = 15$; \quad $2 \times (2)$: $12x - 10y = 34$
\hfill \textbf{A1}

\emph{Both new equations needed, every term multiplied. Common error:
$12x + 9y = 5$ (left side only) earns A0.}

(ii) $(12x + 9y) - (12x - 10y) = 15 - 34$ \hfill \textbf{M1}

$19y = -19$

$y = -1$ \hfill \textbf{A1}

(iii) $4x + 3(-1) = 5$

$4x = 8$

$x = 2$ \hfill \textbf{A1}

\textbf{(b)} $5x + 2y = 3$, \quad $3x + 7y = 25$

$3 \times$ first: $15x + 6y = 9$; \quad $5 \times$ second: $15x + 35y = 125$

Subtracting: $29y = 116$ \hfill \textbf{M1}

$y = 4$ \hfill \textbf{A1}

$5x + 2(4) = 3$, so $5x = -5$ and $x = -1$ \hfill \textbf{A1}

\emph{Accept eliminating $y$ first: $7(5x + 2y) - 2(3x + 7y)$ gives
$29x = -29$.}

%--- Problem 2 [13 marks] ---
\item[2.] [Maximum mark: 13]

\textbf{(a)} $u_{1} + 3d = 17$ \hfill \textbf{A1}

$u_{1} + 9d = 41$ \hfill \textbf{A1}

\textbf{(b)} Subtracting: $6d = 24$ \hfill \textbf{M1}

$d = 4$ \hfill \textbf{A1}

$u_{1} = 17 - 3(4) = 5$ \hfill \textbf{A1}

\emph{Common error: $u_{n} = u_{1} + nd$ gives $u_{1} + 4d = 17$, so
$d = 4$ but $u_{1} = 1$: A0 for $u_{1}$, and FT in (c).}

\textbf{(c)} $S_{20} = \dfrac{20}{2}\left(2(5) + 19(4)\right)$
\hfill \textbf{M1}

$S_{20} = 10(86) = 860$ \hfill \textbf{A1}

\emph{FT from (b). Accept $S_{20} = 10(u_{1} + u_{20}) = 10(5 + 81)$.}

\textbf{(d)} $S_{2} = p(2)^{2} - q(2) = 4p - 2q$ \hfill \textbf{M1}

$4p - 2q = 2$ \hfill \textbf{AG}

\textbf{(e)} $25p - 5q = 35$ \hfill \textbf{A1}

\emph{Accept the simplified form $5p - q = 7$.}

\textbf{(f)} $2p - q = 1$ and $5p - q = 7$

Subtracting: $3p = 6$ \hfill \textbf{M1}

$p = 2$ \hfill \textbf{A1}

$q = 2(2) - 1 = 3$ \hfill \textbf{A1}

\textbf{(g)} $u_{1} = S_{1} = 2(1)^{2} - 3(1) = -1$ \hfill \textbf{A1}

\emph{FT from (f).}

%--- Problem 1 [11 marks] ---
\item[1.] [Maximum mark: 11]

\emph{Elimination is required. Substitution in place of elimination earns
M0 on each elimination step; A marks for correct values still apply.}

\textbf{(a)} $5x + 2y = 3$, \quad $3x + 7y = 25$

$3 \times$ first: $15x + 6y = 9$; \quad $5 \times$ second: $15x + 35y = 125$

Subtracting: $29y = 116$ \hfill \textbf{M1}

$y = 4$ \hfill \textbf{A1}

$5x + 2(4) = 3$, so $x = -1$ \hfill \textbf{A1}

\textbf{(b)} $3x + 2y - z = 1$ \ (1), \quad $2x - 3y + 4z = 19$ \ (2),
\quad $4x + 5y + 2z = 9$ \ (3)

$4 \times (1) + (2)$: $12x + 8y - 4z + 2x - 3y + 4z = 4 + 19$ \hfill \textbf{M1}

$14x + 5y = 23$ \ (4) \hfill \textbf{A1}

$2 \times (1) + (3)$: $6x + 4y - 2z + 4x + 5y + 2z = 2 + 9$ \hfill \textbf{M1}

$10x + 9y = 11$ \ (5) \hfill \textbf{A1}

$9 \times (4)$: $126x + 45y = 207$; \quad $5 \times (5)$: $50x + 45y = 55$

Subtracting: $76x = 152$ \hfill \textbf{M1}

$x = 2$ \hfill \textbf{A1}

$14(2) + 5y = 23$, so $y = -1$ \hfill \textbf{A1}

$z = 3(2) + 2(-1) - 1 = 3$ \hfill \textbf{A1}

\emph{Accept eliminating $x$ or $y$ first; award the same marks for the
two two-variable equations, their solution, and the third value. Common
errors: scaling only part of an equation (for example $2 \times (1)$
written as $6x + 4y - z = 2$) earns M1A0 for that equation; after eliminating
$x$ from (4) and (5), the value found is $y$, not $x$.}

%--- Problem 2 [18 marks] ---
\item[2.] [Maximum mark: 18]

\textbf{(a)} $u_{1} + 2d = 14$ \hfill \textbf{A1}

$\dfrac{8}{2}\left(2u_{1} + 7d\right) = 148$, so $2u_{1} + 7d = 37$
\hfill \textbf{A1}

\textbf{(b)} $2 \times$ first: $2u_{1} + 4d = 28$

Subtracting: $3d = 9$ \hfill \textbf{M1}

$d = 3$ \hfill \textbf{A1}

$u_{1} = 14 - 2(3) = 8$ \hfill \textbf{A1}

\textbf{(c)} $u_{20} = 8 + 19(3)$ \hfill \textbf{M1}

$u_{20} = 65$ \hfill \textbf{A1}

\textbf{(d)} $S_{2}$: $4p - 2q = 2$ \hfill \textbf{A1}

$S_{5}$: $25p - 5q = 35$ \hfill \textbf{A1}

$2p - q = 1$ and $5p - q = 7$; subtracting: $3p = 6$ \hfill \textbf{M1}

$p = 2$, \ $q = 3$ \hfill \textbf{A1A1}

\textbf{(e)} $u_{1} = S_{1} = 2 - 3 = -1$ \hfill \textbf{A1}

$u_{2} = S_{2} - S_{1}$ \hfill \textbf{M1}

$u_{2} = 2 - (-1) = 3$ \hfill \textbf{A1}

$d = 3 - (-1) = 4$ \hfill \textbf{A1}

\emph{Common error: $u_{2} = S_{2} = 2$ earns M0A0, then FT for $d$.}

\textbf{(f)} $u_{n} = -1 + (n - 1)(4)$ \hfill \textbf{M1}

$u_{n} = -1 + 4n - 4$

$u_{n} = 4n - 5$ \hfill \textbf{AG}

\emph{Accept $u_{n} = S_{n} - S_{n-1} = 2n^{2} - 3n - \left(2(n - 1)^{2}
- 3(n - 1)\right)$ for M1, simplified to $4n - 5$.}
