\subsubsection*{1.3 Classwork: Arithmetic Series --- Markscheme}

\begin{enumerate}[itemsep=0.3cm]

%--- Problem 1: S_5 by adding, S_5 by formula, then S_20 [5 marks] ---
\item[1.] [Maximum mark: 5]

The series $3 + 7 + 11 + 15 + \cdots$, so $u_{1} = 3$ and $d = 4$.

\medskip

\textbf{(a)} $S_{5} = 3 + 7 + 11 + 15 + 19 = 55$ \hfill \textbf{A1}

\emph{Accept $55$ obtained by any method, including a sum formula.}

\emph{The sum of only four terms, $3 + 7 + 11 + 15 = 36$, earns no mark.}

\medskip

\textbf{(b)} $S_{5} = \dfrac{5}{2}\bigl(2(3) + (5 - 1)(4)\bigr)$ \hfill \textbf{M1}

$S_{5} = \dfrac{5}{2}(22) = 55$ \hfill \textbf{A1}

\emph{Accept substitution into the second formula,
$S_{5} = \dfrac{5}{2}(3 + 19) = 55$.}

\emph{M1 is for substituting $n = 5$, $u_{1} = 3$ and $d = 4$ (or
$u_{5} = 19$), however the arithmetic is then arranged. The question asks
for the substituted values, so $55$ with no substitution shown earns the
A1 only.}

\medskip

\textbf{(c)} $S_{20} = \dfrac{20}{2}\bigl(2(3) + (20 - 1)(4)\bigr)$ \hfill \textbf{M1}

$S_{20} = 10(6 + 76) = 820$ \hfill \textbf{A1}

\emph{Accept the route $u_{20} = 3 + 19(4) = 79$, then
$S_{20} = \dfrac{20}{2}(3 + 79) = 820$.}

\emph{Using $nd$ in place of $(n - 1)d$, giving $10(6 + 80) = 860$, earns
the M1 only.}

\emph{Accept $820$ from a GDC sum of the first $20$ terms for full marks.
Feedback: write out the formula substitution so the method is visible.}

\newpage

%--- Problem 2: Theatre seating; row 20, total seats, unreserved seats [7 marks] ---
\item[2.] [Maximum mark: 7]

$u_{1} = 18$, $d = 3$, and there are $20$ rows.

\medskip

\textbf{(a)} $u_{20} = 18 + (20 - 1)(3)$ \hfill \textbf{M1}

$u_{20} = 75$ seats \hfill \textbf{A1}

\emph{Using $20$ in place of $(n - 1)$, giving $18 + 20(3) = 78$, earns
the M1 only.}

\medskip

\textbf{(b)} $S_{20} = \dfrac{20}{2}(18 + 75)$ \hfill \textbf{M1}

$S_{20} = 10(93) = 930$ seats \hfill \textbf{A1}

\emph{Accept $S_{20} = \dfrac{20}{2}\bigl(2(18) + 19(3)\bigr) = 930$.}

\emph{FT: when $S_{n} = \dfrac{n}{2}(u_{1} + u_{n})$ is used, award M1A1
for the total correctly obtained from the candidate's own $u_{20}$ in (a).}

\medskip

\textbf{(c)} Seats in the first $8$ rows:

$S_{8} = \dfrac{8}{2}\bigl(2(18) + (8 - 1)(3)\bigr)$ \hfill \textbf{M1}

$S_{8} = 4(57) = 228$ \hfill \textbf{A1}

Seats not reserved: $930 - 228 = 702$ \hfill \textbf{A1}

\emph{Accept the direct route: rows $9$ to $20$ are $12$ rows from
$u_{9} = 42$ to $u_{20} = 75$, so $\dfrac{12}{2}(42 + 75) = 702$. Award M1
for a sum of these $12$ rows, A1 for $u_{9} = 42$, A1 for $702$.}

\emph{Summing the first $9$ rows in place of the first $8$
($S_{9} = 270$, answer $660$) earns the M1 and the final A1 by FT, but not
the A1 for $228$.}

\emph{FT: award the final A1 for the candidate's own total from (b) minus
the candidate's own reserved-row total.}

\end{enumerate}
